# k2p2p Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build k2p2p — a NAT traversal layer that enables peer-to-peer home access, with k2v5 as the data plane after hole punching.

**Architecture:** k2p2p is a standalone top-level package (`k2p2p/`) that orchestrates STUN probing, NAT classification, signaling (central API server), and UDP hole punching (standard + birthday attack). After successful punch, it wraps the resulting QUIC connection as `wire.Dialer` (away side) or `wire.Listener` (home side). Engine treats it identically to a direct k2v5 connection.

**Tech Stack:** Go stdlib, `github.com/pion/stun/v3` (STUN client), `github.com/apernet/quic-go` (QUIC over punched hole), `wire/` (k2v5 framing, auth, k2cc)

**Scope:** This is a large feature. The plan is split into 5 phases, each independently testable and committable. Phases 1-3 are the k2p2p package internals. Phase 4 adds minimal wire/ exports needed by k2p2p. Phase 5 orchestrates everything into Home/Away.

**Spec:** `docs/superpowers/specs/2026-03-24-k2p2p-design.md`

**Blocker resolutions from spec review:**
- BLOCKER-1 (Listen lifetime): Home creates `quic.Transport` on the punched `net.PacketConn`, then calls `QUICServer.Listen()`. Multiple away peers = multiple punch+listen cycles. **Spec amendment:** `Listen()` returns `<-chan wire.Listener` (channel of listeners, one per punched connection), not a single `wire.Listener`.
- BLOCKER-2 (Dial fallback): `away.Dial()` returns `ConnectResult{Dialer, TransportType}` — engine builds `TransportManager` with relay as fallback, same pattern as `wire.RaceResult`.
- BLOCKER-3 (Signaling): Defined in Phase 2 with full API contract. Away uses `POST /p2p/connect` (separate from home's `POST /p2p/register`).
- WARNING (ECH on P2P): Deferred — use standard TLS (not ECH) for punched connections initially. ECH requires a plausible cover server; residential IPs have none. Can add obfuscation later.

// k2p2p/stun.go
package k2p2p

import (
	"context"
	"fmt"
	"net"
	"time"

	"github.com/pion/stun/v3"
)

// stunResult holds the outcome of a single STUN probe.
type stunResult struct {
	reflexive *net.UDPAddr // public-facing address discovered by STUN
	local     *net.UDPAddr // local address used for the probe
	server    string       // STUN server address used
}

// stunProbe sends a STUN Binding Request to the given server and returns
// the reflexive (mapped) address. If conn is nil, a new UDP socket is created.
func stunProbe(ctx context.Context, server string, conn *net.UDPConn) (stunResult, error) {
	var ownConn bool
	if conn == nil {
		c, err := net.ListenUDP("udp4", nil)
		if err != nil {
			return stunResult{}, fmt.Errorf("k2p2p: listen UDP: %w", err)
		}
		conn = c
		ownConn = true
		defer conn.Close()
	}
	_ = ownConn

	serverAddr, err := net.ResolveUDPAddr("udp4", server)
	if err != nil {
		return stunResult{}, fmt.Errorf("k2p2p: resolve STUN server %q: %w", server, err)
	}

	msg, err := stun.Build(stun.TransactionID, stun.BindingRequest)
	if err != nil {
		return stunResult{}, fmt.Errorf("k2p2p: build STUN request: %w", err)
	}

	if deadline, ok := ctx.Deadline(); ok {
		conn.SetDeadline(deadline)
		defer conn.SetDeadline(time.Time{})
	}

	if _, err := conn.WriteToUDP(msg.Raw, serverAddr); err != nil {
		return stunResult{}, fmt.Errorf("k2p2p: send STUN request: %w", err)
	}

	buf := make([]byte, 1500)
	n, _, err := conn.ReadFromUDP(buf)
	if err != nil {
		return stunResult{}, fmt.Errorf("k2p2p: read STUN response: %w", err)
	}

	resp := new(stun.Message)
	resp.Raw = buf[:n]
	if err := resp.Decode(); err != nil {
		return stunResult{}, fmt.Errorf("k2p2p: decode STUN response: %w", err)
	}

	var xorAddr stun.XORMappedAddress
	if err := xorAddr.GetFrom(resp); err != nil {
		return stunResult{}, fmt.Errorf("k2p2p: extract XOR-MAPPED-ADDRESS: %w", err)
	}

	return stunResult{
		reflexive: &net.UDPAddr{IP: xorAddr.IP, Port: xorAddr.Port},
		local:     conn.LocalAddr().(*net.UDPAddr),
		server:    server,
	}, nil
}
```

NAT classification determines whether a peer has "easy" (endpoint-independent mapping = cone) or "hard" (endpoint-dependent = symmetric) NAT. This drives the hole-punch strategy.

**Algorithm:** Send STUN probes from the SAME local socket to TWO different STUN servers. If both return the same reflexive address → easy NAT. If different → hard NAT (symmetric).

// k2p2p/nat.go
package k2p2p

import (
	"context"
	"fmt"
	"log/slog"
	"net"
)

// NATType classifies a peer's NAT behavior.
type NATType int

const (
	NATUnknown NATType = iota
	NATEasy            // Endpoint-independent mapping (full/restricted/port-restricted cone)
	NATHard            // Endpoint-dependent mapping (symmetric)
	NATOpen            // No NAT (reflexive == local)
)

func (t NATType) String() string {
	switch t {
	case NATEasy:
		return "easy"
	case NATHard:
		return "hard"
	case NATOpen:
		return "open"
	default:
		return "unknown"
	}
}

// parseNATType converts a string to NATType.
func parseNATType(s string) NATType {
	switch s {
	case "easy":
		return NATEasy
	case "hard":
		return NATHard
	case "open":
		return NATOpen
	default:
		return NATUnknown
	}
}

// NATResult holds the outcome of NAT classification.
type NATResult struct {
	Type      NATType
	Reflexive *net.UDPAddr // public address (from first successful probe)
	Local     *net.UDPAddr // local socket address used for probing
}

// ClassifyNAT probes multiple STUN servers from the same socket to determine
// NAT type. Requires at least 2 servers. Returns NATEasy if all servers
// return the same reflexive address, NATHard if they differ.
func ClassifyNAT(ctx context.Context, servers []string) (NATResult, error) {
	if len(servers) < 2 {
		return NATResult{}, fmt.Errorf("k2p2p: need at least 2 STUN servers, got %d", len(servers))
	}

	conn, err := net.ListenUDP("udp4", nil)
	if err != nil {
		return NATResult{}, fmt.Errorf("k2p2p: listen UDP: %w", err)
	}
	defer conn.Close()

	var results []stunResult
	for _, server := range servers[:2] {
		r, err := stunProbe(ctx, server, conn)
		if err != nil {
			slog.Debug("k2p2p: STUN probe failed, trying next", "server", server, "err", err)
			continue
		}
		results = append(results, r)
	}

	if len(results) < 2 {
		if len(results) == 1 {
			return NATResult{
				Type:      NATEasy,
				Reflexive: results[0].reflexive,
				Local:     conn.LocalAddr().(*net.UDPAddr),
			}, nil
		}
		return NATResult{}, fmt.Errorf("k2p2p: all STUN servers unreachable")
	}

	local := conn.LocalAddr().(*net.UDPAddr)
	first := results[0].reflexive

	if first.IP.Equal(local.IP) && first.Port == local.Port {
		return NATResult{Type: NATOpen, Reflexive: first, Local: local}, nil
	}

	if first.IP.Equal(results[1].reflexive.IP) && first.Port == results[1].reflexive.Port {
		slog.Info("DIAG: nat-classify", "type", "easy", "reflexive", first.String(), "serversUsed", 2)
		return NATResult{Type: NATEasy, Reflexive: first, Local: local}, nil
	}

	slog.Info("DIAG: nat-classify", "type", "hard", "reflexive", first.String(), "serversUsed", 2)
	return NATResult{Type: NATHard, Reflexive: first, Local: local}, nil
}
```

// k2p2p/signal.go
package k2p2p

import (
	"bytes"
	"context"
	"encoding/json"
	"fmt"
	"net/http"
	"time"
)

// RegisterRequest is sent by home to register its address.
type RegisterRequest struct {
	UDID    string `json:"udid"`
	Addr    string `json:"addr"`     // STUN-discovered public UDP address
	NATType string `json:"nat_type"` // "easy", "hard", "open"
}

// ConnectRequest is sent by away to request connection to a home peer.
// The API server forwards this as a PunchNotify to the home peer.
type ConnectRequest struct {
	HomeUDID string `json:"home_udid"`
	AwayAddr string `json:"away_addr"` // away's STUN-discovered address
	NATType  string `json:"nat_type"`  // away's NAT type
}

// PeerInfo is returned by lookup.
type PeerInfo struct {
	UDID      string    `json:"udid"`
	Addr      string    `json:"addr"`
	NATType   string    `json:"nat_type"`
	UpdatedAt time.Time `json:"updated_at"`
}

// PunchNotify is received by home when away wants to connect.
type PunchNotify struct {
	AwayAddr string `json:"away_addr"`
	NATType  string `json:"nat_type"`
}

// SignalClient communicates with the central API server for peer discovery.
type SignalClient struct {
	baseURL string
	token   string
	http    *http.Client
}

// NewSignalClient creates a signaling client for the given API server.
func NewSignalClient(baseURL, token string) *SignalClient {
	return &SignalClient{
		baseURL: baseURL,
		token:   token,
		http:    &http.Client{Timeout: 10 * time.Second},
	}
}

func (c *SignalClient) doJSON(ctx context.Context, method, path string, body any) (*http.Response, error) {
	var reqBody *bytes.Reader
	if body != nil {
		b, err := json.Marshal(body)
		if err != nil {
			return nil, fmt.Errorf("k2p2p: marshal: %w", err)
		}
		reqBody = bytes.NewReader(b)
	}

	var httpReq *http.Request
	var err error
	if reqBody != nil {
		httpReq, err = http.NewRequestWithContext(ctx, method, c.baseURL+path, reqBody)
	} else {
		httpReq, err = http.NewRequestWithContext(ctx, method, c.baseURL+path, nil)
	}
	if err != nil {
		return nil, fmt.Errorf("k2p2p: build request: %w", err)
	}
	if body != nil {
		httpReq.Header.Set("Content-Type", "application/json")
	}
	httpReq.Header.Set("Authorization", "Bearer "+c.token)

	return c.http.Do(httpReq)
}

// Register reports this peer's STUN-discovered address to the API server.
func (c *SignalClient) Register(ctx context.Context, req RegisterRequest) error {
	resp, err := c.doJSON(ctx, "POST", "/p2p/register", req)
	if err != nil {
		return fmt.Errorf("k2p2p: register: %w", err)
	}
	defer resp.Body.Close()
	if resp.StatusCode != 200 {
		return fmt.Errorf("k2p2p: register: HTTP %d", resp.StatusCode)
	}
	return nil
}

// Lookup queries the API server for a peer's address.
func (c *SignalClient) Lookup(ctx context.Context, udid string) (PeerInfo, error) {
	resp, err := c.doJSON(ctx, "GET", "/p2p/lookup/"+udid, nil)
	if err != nil {
		return PeerInfo{}, fmt.Errorf("k2p2p: lookup: %w", err)
	}
	defer resp.Body.Close()
	if resp.StatusCode == 404 {
		return PeerInfo{}, fmt.Errorf("k2p2p: peer %q not found", udid)
	}
	if resp.StatusCode != 200 {
		return PeerInfo{}, fmt.Errorf("k2p2p: lookup: HTTP %d", resp.StatusCode)
	}
	var info PeerInfo
	if err := json.NewDecoder(resp.Body).Decode(&info); err != nil {
		return PeerInfo{}, fmt.Errorf("k2p2p: decode lookup: %w", err)
	}
	return info, nil
}

// RequestConnect asks the API server to notify a home peer that away
// wants to connect. This is a separate endpoint from Register to avoid
// overwriting home's registration data.
func (c *SignalClient) RequestConnect(ctx context.Context, req ConnectRequest) error {
	resp, err := c.doJSON(ctx, "POST", "/p2p/connect", req)
	if err != nil {
		return fmt.Errorf("k2p2p: connect request: %w", err)
	}
	defer resp.Body.Close()
	if resp.StatusCode != 200 {
		return fmt.Errorf("k2p2p: connect request: HTTP %d", resp.StatusCode)
	}
	return nil
}
```

// ListenNotify opens a WebSocket to the API server and returns a channel
// of PunchNotify messages. The channel is closed when ctx is cancelled
// or the connection drops. Uses safego.Go for panic safety.
func (c *SignalClient) ListenNotify(ctx context.Context, udid string) (<-chan PunchNotify, error) {
	wsURL := c.baseURL + "/p2p/ws?udid=" + udid
	header := http.Header{}
	header.Set("Authorization", "Bearer "+c.token)

	// DialContext returns (*Conn, *http.Response, error) — discard Response.
	conn, _, err := websocket.DefaultDialer.DialContext(ctx, wsURL, header)
	if err != nil {
		return nil, fmt.Errorf("k2p2p: websocket dial: %w", err)
	}

	ch := make(chan PunchNotify, 8)
	safego.Go(func() {
		defer close(ch)
		defer conn.Close()
		for {
			if ctx.Err() != nil {
				return
			}
			_, msg, err := conn.ReadMessage()
			if err != nil {
				return
			}
			var notify PunchNotify
			if err := json.Unmarshal(msg, &notify); err != nil {
				continue
			}
			select {
			case ch <- notify:
			case <-ctx.Done():
				return
			}
		}
	})

	return ch, nil
}

package k2p2p

import (
	"context"
	"fmt"
	"log/slog"
	"net"
	"time"

	"github.com/kaitu-io/k2/safego"
)

const (
	punchMagic    = "k2p2p-punch"
	punchInterval = 100 * time.Millisecond
	punchTimeout  = 5 * time.Second
)

// punchStandard sends UDP punch packets to peerAddr from conn at regular
// intervals while simultaneously listening for incoming punch packets.
// Returns the confirmed peer address on success.
func punchStandard(ctx context.Context, conn *net.UDPConn, peerAddr *net.UDPAddr) (*net.UDPAddr, error) {
	ctx, cancel := context.WithTimeout(ctx, punchTimeout)
	defer cancel()

	// Send punch packets in background.
	safego.Go(func() {
		ticker := time.NewTicker(punchInterval)
		defer ticker.Stop()
		for {
			select {
			case <-ctx.Done():
				return
			case <-ticker.C:
				conn.WriteToUDP([]byte(punchMagic), peerAddr)
			}
		}
	})

	buf := make([]byte, 64)
	for {
		if ctx.Err() != nil {
			return nil, fmt.Errorf("k2p2p: punch timeout: %w", ctx.Err())
		}
		conn.SetReadDeadline(time.Now().Add(punchInterval * 2))
		n, from, err := conn.ReadFromUDP(buf)
		if err != nil {
			if netErr, ok := err.(net.Error); ok && netErr.Timeout() {
				continue
			}
			return nil, fmt.Errorf("k2p2p: punch read: %w", err)
		}
		if n >= len(punchMagic) && string(buf[:len(punchMagic)]) == punchMagic {
			slog.Info("DIAG: punch-ok", "from", from.String(), "method", "standard")
			return from, nil
		}
	}
}

// BirthdayResult holds the outcome of a birthday attack punch.
type BirthdayResult struct {
	Conn     *net.UDPConn // the socket that received a punch reply (caller owns)
	PeerAddr *net.UDPAddr // confirmed peer address
}

// punchBirthday opens multiple UDP sockets on localIP (ports localStart..localEnd)
// and sends punch packets to peerIP (ports peerStart..peerEnd).
// Returns the first socket that receives a valid punch reply.
// If localStart == 0 && localEnd == 0, uses OS-assigned random ports (256 sockets).
func punchBirthday(ctx context.Context, localIP, peerIP net.IP,
	localStart, localEnd, peerStart, peerEnd int) (*BirthdayResult, error) {

	ctx, cancel := context.WithTimeout(ctx, 10*time.Second)
	defer cancel()

	// Determine local ports.
	if localStart == 0 && localEnd == 0 {
		localStart = 0 // OS-assigned
		localEnd = 255  // open 256 sockets
	}
	numLocal := localEnd - localStart + 1

	conns := make([]*net.UDPConn, 0, numLocal)
	for i := 0; i < numLocal; i++ {
		port := 0
		if localStart > 0 {
			port = localStart + i
		}
		c, err := net.ListenUDP("udp4", &net.UDPAddr{IP: localIP, Port: port})
		if err != nil {
			continue
		}
		conns = append(conns, c)
	}
	if len(conns) == 0 {
		return nil, fmt.Errorf("k2p2p: birthday: no local sockets opened")
	}

	type winResult struct {
		conn *net.UDPConn
		peer *net.UDPAddr
	}
	winCh := make(chan winResult, 1)

	// Send punch packets from all sockets to all peer ports.
	safego.Go(func() {
		for round := 0; round < 3; round++ {
			for _, c := range conns {
				for port := peerStart; port <= peerEnd; port++ {
					if ctx.Err() != nil {
						return
					}
					target := &net.UDPAddr{IP: peerIP, Port: port}
					c.WriteToUDP([]byte(punchMagic), target)
				}
			}
			select {
			case <-ctx.Done():
				return
			case <-time.After(200 * time.Millisecond):
			}
		}
	})

	// Listen on all sockets concurrently.
	for _, c := range conns {
		c := c
		safego.Go(func() {
			buf := make([]byte, 64)
			for {
				if ctx.Err() != nil {
					return
				}
				c.SetReadDeadline(time.Now().Add(500 * time.Millisecond))
				n, from, err := c.ReadFromUDP(buf)
				if err != nil {
					if ctx.Err() != nil {
						return
					}
					continue
				}
				if n >= len(punchMagic) && string(buf[:len(punchMagic)]) == punchMagic {
					select {
					case winCh <- winResult{conn: c, peer: from}:
						cancel()
					default:
					}
					return
				}
			}
		})
	}

	// Wait for winner or timeout.
	select {
	case w := <-winCh:
		slog.Info("DIAG: punch-ok", "from", w.peer.String(), "method", "birthday",
			"socketsOpened", len(conns))
		// Close all sockets except the winner.
		for _, c := range conns {
			if c != w.conn {
				c.Close()
			}
		}
		return &BirthdayResult{Conn: w.conn, PeerAddr: w.peer}, nil
	case <-ctx.Done():
		for _, c := range conns {
			c.Close()
		}
		return nil, fmt.Errorf("k2p2p: birthday punch timeout")
	}
}

// selectPunchStrategy returns "standard" or "birthday" based on NAT types.
func selectPunchStrategy(local, remote NATType) string {
	if local == NATHard && remote == NATHard {
		return "birthday"
	}
	return "standard"
}

// PunchResult holds the outcome of a successful hole punch.
type PunchResult struct {
	Conn     *net.UDPConn // the punched socket (caller owns)
	PeerAddr *net.UDPAddr // confirmed peer address
	Strategy string       // "standard" or "birthday"
}

// Punch attempts NAT traversal using the appropriate strategy based on
// local and remote NAT types.
func Punch(ctx context.Context, localNAT, remoteNAT NATType,
	localAddr, peerAddr *net.UDPAddr) (*PunchResult, error) {

	strategy := selectPunchStrategy(localNAT, remoteNAT)
	slog.Info("DIAG: punch-start", "strategy", strategy,
		"localNAT", localNAT.String(), "remoteNAT", remoteNAT.String(),
		"peer", peerAddr.String())

	switch strategy {
	case "birthday":
		portRange := 256
		peerStart := peerAddr.Port - portRange/2
		if peerStart < 1024 {
			peerStart = 1024
		}
		peerEnd := peerStart + portRange - 1
		if peerEnd > 65535 {
			peerEnd = 65535
			peerStart = peerEnd - portRange + 1
		}
		result, err := punchBirthday(ctx, localAddr.IP, peerAddr.IP,
			0, 0, peerStart, peerEnd)
		if err != nil {
			slog.Warn("DIAG: punch-fail", "strategy", strategy, "peer", peerAddr.String(), "err", err)
			return nil, err
		}
		return &PunchResult{Conn: result.Conn, PeerAddr: result.PeerAddr, Strategy: "birthday"}, nil

	default:
		conn, err := net.ListenUDP("udp4", localAddr)
		if err != nil {
			return nil, fmt.Errorf("k2p2p: listen for punch: %w", err)
		}
		confirmed, err := punchStandard(ctx, conn, peerAddr)
		if err != nil {
			conn.Close()
			slog.Warn("DIAG: punch-fail", "strategy", strategy, "peer", peerAddr.String(), "err", err)
			return nil, err
		}
		return &PunchResult{Conn: conn, PeerAddr: confirmed, Strategy: "standard"}, nil
	}
}

// wire/quic_p2p.go
//
// Factory functions for creating QUICClient/QUICServer from pre-established
// connections (k2p2p NAT traversal). These stay in the wire package to access
// unexported fields while respecting encapsulation — external packages use
// these functions, never direct field writes.
package wire

import (
	"github.com/apernet/quic-go"
	"github.com/kaitu-io/k2/wire/k2cc"
)

// NewQUICClientFromConn creates a QUICClient backed by an existing QUIC
// connection (e.g., from NAT-traversed hole punch). Initializes K2CC
// congestion control, session allocator, and UDP mux — the same setup
// that QUICClient.connect() does at wire/quic.go:453-472.
//
// The transport parameter is the quic.Transport that owns the underlying
// UDP socket. The client takes ownership and will close it on Close().
func NewQUICClientFromConn(conn *quic.Conn, transport *quic.Transport, cfg Config, store *TicketStore) *QUICClient {
	c := NewQUICClient(cfg, store)

	// Set K2CC congestion control (same as connect() at quic.go:454)
	sender := k2cc.NewCCSenderGFWMode(k2cc.DefaultMaxRate)
	conn.SetCongestionControl(sender)

	c.mu.Lock()
	c.conn = conn
	c.transport = transport
	c.ccSender = sender
	c.alloc = NewSessionIDAllocator()
	c.udpMux = newQuicUDPMux(conn, &c.portHopAttempts, &c.portHopExhaustedAt)
	c.mu.Unlock()

	// Start accept loop for server-initiated overflow streams
	// (same as connect() at quic.go:474)
	go c.acceptOverflowStreams(conn, c.udpMux)

	return c
}

// NewQUICServerFromListener creates a QUICServer and starts accepting
// connections from the given listener. Convenience wrapper combining
// NewQUICServer() + Listen() — the same pattern as server/server.go:352.
func NewQUICServerFromListener(ln *quic.Listener, store *TicketStore) *QUICServer {
	s := NewQUICServer(store)
	s.Listen(ln)
	return s
}
```

- The encapsulation rule says "call its methods" — these ARE methods on the owning struct's package
- External packages (k2p2p/) call only the exported `NewQUICClientFromConn`

// k2p2p/away.go
package k2p2p

import (
	"context"
	"crypto/tls"
	"fmt"
	"log/slog"
	"net"
	"time"

	"github.com/apernet/quic-go"
	"github.com/kaitu-io/k2/wire"
)

// ConnectResult is the output of a successful Dial().
// Engine uses this to build TransportManager (same pattern as wire.RaceResult).
type ConnectResult struct {
	Dialer        wire.Dialer
	TransportType string // "p2p-direct" or "p2p-relay"
}

// Dial connects to a home peer via NAT traversal.
// Steps: STUN probe → classify NAT → lookup home → signal connect → punch → QUIC dial.
func Dial(ctx context.Context, cfg Config) (*ConnectResult, error) {
	stunServers := cfg.STUNServers
	if len(stunServers) == 0 {
		stunServers = DefaultSTUNServers()
	}

	// 1. STUN probe + NAT classification.
	natResult, err := ClassifyNAT(ctx, stunServers)
	if err != nil {
		return nil, fmt.Errorf("k2p2p: away: NAT classification: %w", err)
	}
	slog.Info("DIAG: nat-classify",
		"type", natResult.Type.String(),
		"reflexive", natResult.Reflexive.String(),
		"serversUsed", 2)

	// 2. Lookup home peer.
	signal := NewSignalClient(cfg.SignalServer, cfg.Token)
	homeInfo, err := signal.Lookup(ctx, cfg.HomeUDID)
	if err != nil {
		return nil, fmt.Errorf("k2p2p: away: lookup home: %w", err)
	}

	homeAddr, err := net.ResolveUDPAddr("udp4", homeInfo.Addr)
	if err != nil {
		return nil, fmt.Errorf("k2p2p: away: resolve home addr: %w", err)
	}

	// 3. Signal connect request (so API notifies home with our address).
	err = signal.RequestConnect(ctx, ConnectRequest{
		HomeUDID: cfg.HomeUDID,
		AwayAddr: natResult.Reflexive.String(),
		NATType:  natResult.Type.String(),
	})
	if err != nil {
		slog.Warn("k2p2p: away: connect request failed (proceeding anyway)", "err", err)
	}

	// 4. Punch.
	remoteNAT := parseNATType(homeInfo.NATType)
	punchResult, err := Punch(ctx, natResult.Type, remoteNAT, natResult.Local, homeAddr)
	if err != nil {
		slog.Warn("DIAG: punch-fail", "peer", homeAddr.String(), "err", err)
		// TODO: fallback to relay via cfg.RelayServer
		return nil, fmt.Errorf("k2p2p: away: punch failed: %w", err)
	}

	// 5. QUIC dial over punched hole.
	dialer, err := dialQUICOverPunch(ctx, punchResult, cfg)
	if err != nil {
		punchResult.Conn.Close()
		return nil, fmt.Errorf("k2p2p: away: QUIC dial: %w", err)
	}

	slog.Info("DIAG: p2p-connected",
		"peer", punchResult.PeerAddr.String(),
		"method", punchResult.Strategy,
		"dialMs", time.Since(time.Now()).Milliseconds())

	return &ConnectResult{
		Dialer:        dialer,
		TransportType: "p2p-direct",
	}, nil
}

// dialQUICOverPunch establishes a QUIC connection over a punched UDP socket.
// Mirrors the setup in wire/quic.go:connect() (lines 407-476):
//   1. Create quic.Transport from existing UDP conn
//   2. Build TLS config (standard, no ECH for P2P — residential IPs have no cover story)
//   3. Dial QUIC with DefaultQUICConfig
//   4. Wrap as QUICClient with K2CC, alloc, udpMux via NewQUICClientFromConn
func dialQUICOverPunch(ctx context.Context, punch *PunchResult, cfg Config) (wire.Dialer, error) {
	// 1. quic.Transport wraps the punched UDP socket (same pattern as quic.go:417)
	tr := &quic.Transport{Conn: punch.Conn}

	// 2. TLS config — standard TLS, no ECH (P2P residential IPs)
	tlsCfg := &tls.Config{
		NextProtos:         []string{"h3"},
		ServerName:         cfg.WireCfg.Host,
		InsecureSkipVerify: cfg.WireCfg.Insecure,
	}
	// If pin is set, add pin verification (same as wire.BuildClientTLSConfig)
	if cfg.WireCfg.Pin != "" {
		// TODO: integrate pin verification from wire.BuildClientTLSConfig
		// For now, rely on InsecureSkipVerify for dev/testing
	}

	// 3. QUIC dial with standard config (quic.go:433-436)
	quicCfg := wire.DefaultQUICConfig()
	conn, err := tr.Dial(ctx, punch.PeerAddr, tlsCfg, quicCfg)
	if err != nil {
		tr.Close()
		return nil, fmt.Errorf("k2p2p: QUIC dial: %w", err)
	}

	// 4. Wrap as QUICClient — K2CC + alloc + udpMux initialized inside
	// (mirrors connect() at quic.go:454-474)
	return wire.NewQUICClientFromConn(conn, tr, cfg.WireCfg, cfg.TicketStore), nil
}

// k2p2p/home.go
package k2p2p

import (
	"context"
	"crypto/tls"
	"fmt"
	"log/slog"
	"net"
	"time"

	"github.com/apernet/quic-go"
	"github.com/sasha-s/go-deadlock"

	"github.com/kaitu-io/k2/safego"
	"github.com/kaitu-io/k2/wire"
)

// Listen starts the home peer: registers with signal server, waits for
// away peers, punches holes, and accepts QUIC connections.
//
// Returns a channel of wire.Listener — one per successfully punched connection.
// The caller should run server proxy logic on each Listener.
// Channel is closed when ctx is cancelled.
//
// Spec amendment: returns <-chan wire.Listener instead of single wire.Listener
// because each away peer requires a separate punch+QUIC accept cycle.
func Listen(ctx context.Context, cfg Config) (<-chan wire.Listener, error) {
	stunServers := cfg.STUNServers
	if len(stunServers) == 0 {
		stunServers = DefaultSTUNServers()
	}

	// 1. STUN probe + NAT classification.
	natResult, err := ClassifyNAT(ctx, stunServers)
	if err != nil {
		return nil, fmt.Errorf("k2p2p: home: NAT classification: %w", err)
	}
	slog.Info("DIAG: nat-classify",
		"type", natResult.Type.String(),
		"reflexive", natResult.Reflexive.String(),
		"serversUsed", 2)

	// 2. Register with signal server.
	signal := NewSignalClient(cfg.SignalServer, cfg.Token)
	err = signal.Register(ctx, RegisterRequest{
		UDID:    cfg.HomeUDID,
		Addr:    natResult.Reflexive.String(),
		NATType: natResult.Type.String(),
	})
	if err != nil {
		return nil, fmt.Errorf("k2p2p: home: register: %w", err)
	}

	// Protect natResult from concurrent access (keepalive writes, punch reads).
	var natMu deadlock.Mutex
	getNAT := func() NATResult {
		natMu.Lock()
		defer natMu.Unlock()
		return natResult
	}

	// 3. Keepalive: re-register every 25s to refresh NAT mapping + API entry.
	safego.Go(func() {
		ticker := time.NewTicker(25 * time.Second)
		defer ticker.Stop()
		for {
			select {
			case <-ctx.Done():
				return
			case <-ticker.C:
				if nr, err := ClassifyNAT(ctx, stunServers); err == nil {
					natMu.Lock()
					natResult = nr
					natMu.Unlock()
				}
				nr := getNAT()
				signal.Register(ctx, RegisterRequest{
					UDID:    cfg.HomeUDID,
					Addr:    nr.Reflexive.String(),
					NATType: nr.Type.String(),
				})
			}
		}
	})

	// 4. Listen for punch notifications.
	notifyCh, err := signal.ListenNotify(ctx, cfg.HomeUDID)
	if err != nil {
		return nil, fmt.Errorf("k2p2p: home: listen notify: %w", err)
	}

	listenerCh := make(chan wire.Listener, 4)

	safego.Go(func() {
		defer close(listenerCh)
		for {
			select {
			case <-ctx.Done():
				return
			case notify, ok := <-notifyCh:
				if !ok {
					return
				}
				safego.Go(func() {
					nat := getNAT()
					result, err := handlePunchRequest(ctx, nat, notify)
					if err != nil {
						slog.Warn("DIAG: punch-fail", "away", notify.AwayAddr, "err", err)
						return
					}

					// Create QUIC listener on the punched UDP socket.
					// Same pattern as server/server.go:342-352:
					//   quicTransport := &quic.Transport{Conn: udpConn}
					//   quicLn := quicTransport.Listen(tlsCfg, wire.ServerQUICConfig())
					//   quicSrv.Listen(quicLn)
					tr := &quic.Transport{Conn: result.Conn}

					// Server TLS config. Home needs a TLS cert to accept QUIC.
					// Use wire.GenerateCertPair() for self-signed certs (same as
					// k2s auto-provisioning at server/server.go setupAutoProvision).
					// Away side must use insecure=1 or pin= to connect.
					//
					// If cfg.TLSCert is provided (from persistent storage), use it.
					// Otherwise generate ephemeral certs on the fly.
					var tlsCert tls.Certificate
					if cfg.TLSCert != nil {
						tlsCert = *cfg.TLSCert
					} else {
						certPair, certErr := wire.GenerateCertPair()
						if certErr != nil {
							slog.Warn("k2p2p: home: generate cert failed", "err", certErr)
							result.Conn.Close()
							return
						}
						tlsCert = certPair.EC // use EC cert (lighter, faster handshake)
					}
					tlsCfg := &tls.Config{
						NextProtos:   []string{"h3"},
						Certificates: []tls.Certificate{tlsCert},
					}

					quicLn, err := tr.Listen(tlsCfg, wire.ServerQUICConfig())
					if err != nil {
						slog.Warn("k2p2p: home: QUIC listen failed", "err", err)
						result.Conn.Close()
						return
					}

					listener := wire.NewQUICServerFromListener(quicLn, cfg.TicketStore)

					slog.Info("DIAG: p2p-connected",
						"peer", result.PeerAddr.String(),
						"method", result.Strategy)

					select {
					case listenerCh <- listener:
					case <-ctx.Done():
						listener.Close()
					}
				})
			}
		}
	})

	return listenerCh, nil
}

// handlePunchRequest processes a single away peer connection attempt.
// Returns the PunchResult (punched UDP socket) on success.
func handlePunchRequest(ctx context.Context, localNAT NATResult, notify PunchNotify) (*PunchResult, error) {
	awayAddr, err := net.ResolveUDPAddr("udp4", notify.AwayAddr)
	if err != nil {
		return nil, fmt.Errorf("k2p2p: resolve away addr: %w", err)
	}

	remoteNAT := parseNATType(notify.NATType)
	return Punch(ctx, localNAT.Type, remoteNAT, localNAT.Local, awayAddr)
}
