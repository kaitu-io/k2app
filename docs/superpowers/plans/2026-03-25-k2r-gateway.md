//go:build linux

package gateway

import (
    "fmt"
    "net"
    "os/exec"
    "strings"
)

// Interceptor installs/removes firewall rules that redirect LAN traffic
// to a local TPROXY listener.
type Interceptor interface {
    // Install sets up firewall rules. Must be called after resolving all exclude IPs.
    Install(cfg InterceptConfig) error
    // Remove cleans up all firewall rules (idempotent, errors swallowed).
    Remove()
    // Name returns the backend name ("nftables" or "iptables").
    Name() string
}

// InterceptConfig describes what traffic to intercept.
type InterceptConfig struct {
    ListenPort int      // TPROXY listen port (default 12345)
    LANSubnets []string // LAN subnets to intercept (e.g. "192.168.1.0/24")
    ExcludeIPs []string // IPs to exclude from interception
}

// DefaultListenPort is the local TPROXY listener port.
const DefaultListenPort = 12345

// NewInterceptor auto-detects and returns the best available firewall backend.
// Prefers nftables, falls back to iptables.
func NewInterceptor() (Interceptor, error) {
    if _, err := exec.LookPath("nft"); err == nil {
        return &nftInterceptor{}, nil
    }
    if _, err := exec.LookPath("iptables"); err == nil {
        return &iptInterceptor{}, nil
    }
    return nil, fmt.Errorf("gateway: neither nft nor iptables found")
}

// splitIPsByFamily separates IPs into IPv4 and IPv6 lists.
func splitIPsByFamily(ips []string) (v4, v6 []string) {
    for _, ip := range ips {
        parsed := net.ParseIP(ip)
        if parsed == nil {
            continue
        }
        if parsed.To4() != nil {
            v4 = append(v4, ip)
        } else {
            v6 = append(v6, ip)
        }
    }
    return
}

// splitSubnetsByFamily separates CIDR subnets into IPv4 and IPv6.
func splitSubnetsByFamily(subnets []string) (v4, v6 []string) {
    for _, s := range subnets {
        ip, _, err := net.ParseCIDR(s)
        if err != nil {
            continue
        }
        if ip.To4() != nil {
            v4 = append(v4, s)
        } else {
            v6 = append(v6, s)
        }
    }
    return
}

// runCmd runs a command and returns an error with output on failure.
func runCmd(name string, args ...string) error {
    cmd := exec.Command(name, args...)
    if out, err := cmd.CombinedOutput(); err != nil {
        return fmt.Errorf("%s %s: %s: %w", name, strings.Join(args, " "), out, err)
    }
    return nil
}

//go:build linux

package gateway

import (
    "fmt"
    "log/slog"
    "os/exec"
    "strings"
)

// nftInterceptor implements Interceptor using nftables.
// Uses a dedicated table "inet k2r" (atomic create/delete).
type nftInterceptor struct {
    installed bool
}

func (n *nftInterceptor) Name() string { return "nftables" }

func (n *nftInterceptor) Install(cfg InterceptConfig) error {
    script := n.buildScript(cfg)

    cmd := exec.Command("nft", "-f", "-")
    cmd.Stdin = strings.NewReader(script)
    if out, err := cmd.CombinedOutput(); err != nil {
        return fmt.Errorf("nft script: %s: %w", out, err)
    }

    // IP routing rules for marked packets.
    _ = runCmd("ip", "rule", "add", "fwmark", "1", "table", "100")
    _ = runCmd("ip", "route", "add", "local", "0.0.0.0/0", "dev", "lo", "table", "100")
    _ = runCmd("ip", "-6", "rule", "add", "fwmark", "1", "table", "100")
    _ = runCmd("ip", "-6", "route", "add", "local", "::/0", "dev", "lo", "table", "100")

    n.installed = true
    slog.Info("DIAG: gw-intercept-install", "backend", "nftables",
        "subnets", cfg.LANSubnets, "excludeCount", len(cfg.ExcludeIPs))
    return nil
}

func (n *nftInterceptor) Remove() {
    _ = runCmd("nft", "delete", "table", "inet", "k2r")
    _ = runCmd("ip", "rule", "del", "fwmark", "1", "table", "100")
    _ = runCmd("ip", "route", "del", "local", "0.0.0.0/0", "dev", "lo", "table", "100")
    _ = runCmd("ip", "-6", "rule", "del", "fwmark", "1", "table", "100")
    _ = runCmd("ip", "-6", "route", "del", "local", "::/0", "dev", "lo", "table", "100")
    n.installed = false
    slog.Info("gateway: nftables rules cleaned up")
}

// buildScript generates the atomic nftables script.
// Handles both IPv4 and IPv6 subnets and excludes.
func (n *nftInterceptor) buildScript(cfg InterceptConfig) string {
    port := fmt.Sprintf(":%d", cfg.ListenPort)
    v4Subnets, v6Subnets := splitSubnetsByFamily(cfg.LANSubnets)
    v4Excludes, v6Excludes := splitIPsByFamily(cfg.ExcludeIPs)

    var lines []string
    lines = append(lines, "add table inet k2r")

    // Exclude sets (only if non-empty).
    if len(v4Excludes) > 0 {
        lines = append(lines, fmt.Sprintf(
            "add set inet k2r exclude_v4 { type ipv4_addr ; elements = { %s } ; }",
            strings.Join(v4Excludes, ", ")))
    }
    if len(v6Excludes) > 0 {
        lines = append(lines, fmt.Sprintf(
            "add set inet k2r exclude_v6 { type ipv6_addr ; elements = { %s } ; }",
            strings.Join(v6Excludes, ", ")))
    }

    // Prerouting chain.
    lines = append(lines,
        "add chain inet k2r prerouting { type filter hook prerouting priority mangle ; policy accept ; }")
    lines = append(lines,
        `add rule inet k2r prerouting iifname "lo" return`)

    // Exclude rules (only if sets exist).
    if len(v4Excludes) > 0 {
        lines = append(lines,
            "add rule inet k2r prerouting ip daddr @exclude_v4 return")
    }
    if len(v6Excludes) > 0 {
        lines = append(lines,
            "add rule inet k2r prerouting ip6 daddr @exclude_v6 return")
    }

    // IPv4 TPROXY rules per LAN subnet.
    for _, subnet := range v4Subnets {
        lines = append(lines, fmt.Sprintf(
            "add rule inet k2r prerouting ip saddr %s meta l4proto tcp tproxy to %s meta mark set 1",
            subnet, port))
        lines = append(lines, fmt.Sprintf(
            "add rule inet k2r prerouting ip saddr %s meta l4proto udp tproxy to %s meta mark set 1",
            subnet, port))
    }

    // IPv6 TPROXY rules per LAN subnet.
    for _, subnet := range v6Subnets {
        lines = append(lines, fmt.Sprintf(
            "add rule inet k2r prerouting ip6 saddr %s meta l4proto tcp tproxy to %s meta mark set 1",
            subnet, port))
        lines = append(lines, fmt.Sprintf(
            "add rule inet k2r prerouting ip6 saddr %s meta l4proto udp tproxy to %s meta mark set 1",
            subnet, port))
    }

    return strings.Join(lines, "\n") + "\n"
}

// iptInterceptor implements Interceptor using iptables/ip6tables.
// Uses a custom chain K2R in the mangle table.
type iptInterceptor struct {
    installed    bool
    hadV4Subnet bool
    hadV6Subnet bool
    v4Subnets   []string
    v6Subnets   []string
}

func (i *iptInterceptor) Name() string { return "iptables" }

// cmdSpec holds a command name and its arguments (for testability).
type cmdSpec struct {
    name string
    args []string
}

// Install is defined in the updated block below (after buildIptCommands).

func (i *iptInterceptor) Remove() {
    // IPv4 cleanup.
    if i.hadV4Subnet {
        for _, s := range i.v4Subnets {
            _ = runCmd("iptables", "-t", "mangle", "-D", "PREROUTING", "-s", s, "-j", "K2R")
        }
        _ = runCmd("iptables", "-t", "mangle", "-F", "K2R")
        _ = runCmd("iptables", "-t", "mangle", "-X", "K2R")
    }
    // IPv6 cleanup.
    if i.hadV6Subnet {
        for _, s := range i.v6Subnets {
            _ = runCmd("ip6tables", "-t", "mangle", "-D", "PREROUTING", "-s", s, "-j", "K2R")
        }
        _ = runCmd("ip6tables", "-t", "mangle", "-F", "K2R")
        _ = runCmd("ip6tables", "-t", "mangle", "-X", "K2R")
    }
    // IP routing rules.
    _ = runCmd("ip", "rule", "del", "fwmark", "1", "table", "100")
    _ = runCmd("ip", "route", "del", "local", "0.0.0.0/0", "dev", "lo", "table", "100")
    _ = runCmd("ip", "-6", "rule", "del", "fwmark", "1", "table", "100")
    _ = runCmd("ip", "-6", "route", "del", "local", "::/0", "dev", "lo", "table", "100")

    slog.Info("gateway: iptables rules cleaned up")

// buildInstallCommands is a pure function — generates commands without mutating receiver.
// State for cleanup is saved separately in Install() after successful execution.
func buildIptCommands(cfg InterceptConfig) (cmds []cmdSpec, v4Subnets, v6Subnets []string) {
    v4Subnets, v6Subnets = splitSubnetsByFamily(cfg.LANSubnets)
    v4Excludes, v6Excludes := splitIPsByFamily(cfg.ExcludeIPs)
    port := fmt.Sprintf("%d", cfg.ListenPort)

    // IPv4 rules.
    if len(v4Subnets) > 0 {
        cmds = append(cmds, cmdSpec{"iptables", []string{"-t", "mangle", "-N", "K2R"}})
        for _, ip := range v4Excludes {
            cmds = append(cmds, cmdSpec{"iptables", []string{"-t", "mangle", "-A", "K2R", "-d", ip, "-j", "RETURN"}})
        }
        cmds = append(cmds,
            cmdSpec{"iptables", []string{"-t", "mangle", "-A", "K2R", "-p", "tcp", "-j", "TPROXY", "--on-port", port, "--tproxy-mark", "1"}},
            cmdSpec{"iptables", []string{"-t", "mangle", "-A", "K2R", "-p", "udp", "-j", "TPROXY", "--on-port", port, "--tproxy-mark", "1"}},
        )
        for _, s := range v4Subnets {
            cmds = append(cmds, cmdSpec{"iptables", []string{"-t", "mangle", "-A", "PREROUTING", "-s", s, "-j", "K2R"}})
        }
    }

    // IPv6 rules.
    if len(v6Subnets) > 0 {
        cmds = append(cmds, cmdSpec{"ip6tables", []string{"-t", "mangle", "-N", "K2R"}})
        for _, ip := range v6Excludes {
            cmds = append(cmds, cmdSpec{"ip6tables", []string{"-t", "mangle", "-A", "K2R", "-d", ip, "-j", "RETURN"}})
        }
        cmds = append(cmds,
            cmdSpec{"ip6tables", []string{"-t", "mangle", "-A", "K2R", "-p", "tcp", "-j", "TPROXY", "--on-port", port, "--tproxy-mark", "1"}},
            cmdSpec{"ip6tables", []string{"-t", "mangle", "-A", "K2R", "-p", "udp", "-j", "TPROXY", "--on-port", port, "--tproxy-mark", "1"}},
        )
        for _, s := range v6Subnets {
            cmds = append(cmds, cmdSpec{"ip6tables", []string{"-t", "mangle", "-A", "PREROUTING", "-s", s, "-j", "K2R"}})
        }
    }

    // IP routing rules.
    cmds = append(cmds,
        cmdSpec{"ip", []string{"rule", "add", "fwmark", "1", "table", "100"}},
        cmdSpec{"ip", []string{"route", "add", "local", "0.0.0.0/0", "dev", "lo", "table", "100"}},

        cmdSpec{"ip", []string{"-6", "route", "add", "local", "::/0", "dev", "lo", "table", "100"}},
    )

    return cmds, v4Subnets, v6Subnets

// DiscoverLANSubnets finds private subnets on physical interfaces.
// Returns CIDR strings like "192.168.1.0/24".
func DiscoverLANSubnets() ([]string, error) {
    ifaces, err := net.Interfaces()
    if err != nil {
        return nil, err
    }

    var subnets []string
    for _, iface := range ifaces {
        if iface.Flags&net.FlagLoopback != 0 {
            continue
        }
        if iface.Flags&net.FlagUp == 0 {
            continue
        }
        if isVirtualInterface(iface.Name) {
            continue
        }

        addrs, _ := iface.Addrs()
        for _, addr := range addrs {
            ipNet, ok := addr.(*net.IPNet)
            if !ok {
                continue
            }
            if !ipNet.IP.IsPrivate() {
                continue
            }
            subnets = append(subnets, ipNet.String())
        }
    }
    if len(subnets) == 0 {
        return nil, errors.New("gateway: no private LAN subnets found")
    }
    return subnets, nil
}

// isVirtualInterface returns true for non-physical interfaces that
// should not be intercepted (Docker, veth, TUN, etc.).
// Note: "br-lan" (OpenWRT LAN bridge) is NOT filtered — it carries real LAN traffic.
func isVirtualInterface(name string) bool {
    prefixes := []string{"docker", "veth", "tun", "tap", "virbr", "lo"}
    for _, p := range prefixes {
        if strings.HasPrefix(name, p) {
            return true
        }
    }
    return false
}

// resolveExcludeIPs builds the list of IPs that must NOT be intercepted.
// Includes: k2s server IPs, direct DNS upstream IPs, gateway's own host IPs.
func resolveExcludeIPs(serverURL string, directDNS []string) ([]string, error) {
    // 1. Extract hostname from k2v5:// URL.
    host := extractServerHost(serverURL)
    if host == "" {
        return nil, fmt.Errorf("gateway: cannot extract host from server URL")
    }

    // 2. Resolve server hostname to IPs.
    serverIPs, err := net.LookupHost(host)
    if err != nil {
        return nil, fmt.Errorf("resolve server %s: %w", host, err)
    }

    exclude := make([]string, 0, len(serverIPs)+len(directDNS)+8)
    exclude = append(exclude, serverIPs...)

    // 3. Direct DNS upstream IPs (prevent query loops).
    for _, dns := range directDNS {
        h, _, _ := net.SplitHostPort(dns)
        if h == "" {
            h = dns
        }
        if ip := net.ParseIP(h); ip != nil {
            exclude = append(exclude, ip.String())
        }
    }

    // 4. Gateway's own host IPs on all non-loopback interfaces.
    ifaces, _ := net.Interfaces()
    for _, iface := range ifaces {
        if iface.Flags&net.FlagLoopback != 0 {
            continue
        }
        addrs, _ := iface.Addrs()
        for _, addr := range addrs {
            ipNet, ok := addr.(*net.IPNet)
            if !ok {
                continue
            }
            exclude = append(exclude, ipNet.IP.String())
        }
    }

    return exclude, nil
}

// extractServerHost extracts the hostname from a k2v5:// URL.
// Example: "k2v5://user:pass@example.com:443/path" → "example.com"
func extractServerHost(serverURL string) string {
    // k2v5:// uses standard URL format. net/url handles this.
    // But k2v5:// scheme may confuse net/url. Use manual parsing.
    // Format: k2v5://[userinfo@]host[:port]/path[?query]
    s := serverURL
    if idx := strings.Index(s, "://"); idx >= 0 {
        s = s[idx+3:]
    }
    // Strip userinfo.
    if idx := strings.Index(s, "@"); idx >= 0 {
        s = s[idx+1:]
    }
    // Strip path.
    if idx := strings.Index(s, "/"); idx >= 0 {
        s = s[:idx]
    }
    // Strip port.
    host, _, err := net.SplitHostPort(s)
    if err != nil {
        return s // no port
    }
    return host
}

//go:build linux

package gateway

import (
    "context"
    "fmt"
    "net"
    "syscall"
    "time"
)

// newTPROXYTCPListener creates a TCP listener with IP_TRANSPARENT.
// When TPROXY redirects a connection, conn.LocalAddr() returns
// the original destination address (preserved by the kernel).
func newTPROXYTCPListener(ctx context.Context, port int) (net.Listener, error) {
    lc := net.ListenConfig{
        Control: func(network, address string, c syscall.RawConn) error {
            return c.Control(func(fd uintptr) {
                syscall.SetsockoptInt(int(fd), syscall.SOL_IP, syscall.IP_TRANSPARENT, 1)
            })
        },
    }
    return lc.Listen(ctx, "tcp", fmt.Sprintf("0.0.0.0:%d", port))
}

// newTPROXYUDPListener creates a UDP listener with IP_TRANSPARENT
// and IP_RECVORIGDSTADDR. Original destination is extracted from
// ancillary data in readUDPOrigDst().
func newTPROXYUDPListener(ctx context.Context, port int) (*net.UDPConn, error) {
    lc := net.ListenConfig{
        Control: func(network, address string, c syscall.RawConn) error {
            return c.Control(func(fd uintptr) {
                syscall.SetsockoptInt(int(fd), syscall.SOL_IP, syscall.IP_TRANSPARENT, 1)
                syscall.SetsockoptInt(int(fd), syscall.SOL_IP, syscall.IP_RECVORIGDSTADDR, 1)
            })
        },
    }
    conn, err := lc.ListenPacket(ctx, "udp", fmt.Sprintf("0.0.0.0:%d", port))
    if err != nil {
        return nil, err
    }
    return conn.(*net.UDPConn), nil
}

// readUDPOrigDst reads a UDP packet and extracts the original destination
// from the IP_RECVORIGDSTADDR ancillary message.
//
// struct sockaddr_in layout: family(2) + port(2) + addr(4)
// IPv6 (IPV6_RECVORIGDSTADDR): family(2) + port(2) + flowinfo(4) + addr(16) + scope(4)
func readUDPOrigDst(conn *net.UDPConn, buf []byte) (n int, src, dst *net.UDPAddr, err error) {
    oob := make([]byte, 128) // enough for both IPv4 and IPv6 ancillary data
    n, oobn, _, srcAddr, err := conn.ReadMsgUDP(buf, oob)
    if err != nil {
        return 0, nil, nil, err
    }

    msgs, err := syscall.ParseSocketControlMessage(oob[:oobn])
    if err != nil {
        return 0, nil, nil, fmt.Errorf("parse control message: %w", err)
    }

    for _, msg := range msgs {
        // IPv4: SOL_IP / IP_RECVORIGDSTADDR
        if msg.Header.Level == syscall.SOL_IP && msg.Header.Type == syscall.IP_RECVORIGDSTADDR {
            if len(msg.Data) >= 8 {
                port := int(msg.Data[2])<<8 | int(msg.Data[3])
                ip := net.IPv4(msg.Data[4], msg.Data[5], msg.Data[6], msg.Data[7])
                return n, srcAddr, &net.UDPAddr{IP: ip, Port: port}, nil
            }
        }
        // IPv6: SOL_IPV6 / IPV6_RECVORIGDSTADDR (20 = IPV6_RECVORIGDSTADDR on Linux)
        if msg.Header.Level == syscall.SOL_IPV6 && msg.Header.Type == 74 { // IPV6_RECVORIGDSTADDR
            if len(msg.Data) >= 28 {
                port := int(msg.Data[2])<<8 | int(msg.Data[3])
                ip := make(net.IP, 16)
                copy(ip, msg.Data[8:24])
                return n, srcAddr, &net.UDPAddr{IP: ip, Port: port}, nil
            }
        }
    }

    return n, srcAddr, srcAddr, nil // fallback
}

// tproxyUDPConn wraps a single UDP packet as a net.PacketConn.
// First ReadFrom returns the initial packet; subsequent reads block until Close.
// WriteTo sends responses back to the original source via the shared UDP socket.
type tproxyUDPConn struct {
    conn    *net.UDPConn // shared listener socket
    src     *net.UDPAddr // original source (LAN device)
    dst     *net.UDPAddr // original destination
    initial []byte       // first packet data (copied)
    read    bool
    done    chan struct{}
}

func newTProxyUDPConn(conn *net.UDPConn, src, dst *net.UDPAddr, initial []byte) *tproxyUDPConn {
    data := make([]byte, len(initial))
    copy(data, initial)
    return &tproxyUDPConn{
        conn:    conn,
        src:     src,
        dst:     dst,
        initial: data,
        done:    make(chan struct{}),
    }
}

func (c *tproxyUDPConn) ReadFrom(b []byte) (int, net.Addr, error) {
    if !c.read {
        c.read = true
        n := copy(b, c.initial)
        return n, c.src, nil
    }
    <-c.done
    return 0, nil, net.ErrClosed
}

// WriteTo sends the response back to the original LAN source address.
// The addr parameter is intentionally ignored — TPROXY wraps each UDP packet
// as a single-session conn, so all responses go to the originating LAN device.
// This matches the existing tproxy pattern in provider/tproxy_linux.go.
func (c *tproxyUDPConn) WriteTo(b []byte, _ net.Addr) (int, error) {
    return c.conn.WriteTo(b, c.src)
}

func (c *tproxyUDPConn) Close() error {
    select {
    case <-c.done:
    default:
        close(c.done)
    }
    return nil
}

func (c *tproxyUDPConn) LocalAddr() net.Addr               { return c.dst }
func (c *tproxyUDPConn) SetDeadline(t time.Time) error      { return nil }
func (c *tproxyUDPConn) SetReadDeadline(t time.Time) error   { return nil }
func (c *tproxyUDPConn) SetWriteDeadline(t time.Time) error  { return nil }

// gatewayProvider implements provider.Provider using TPROXY.
// Owns the Interceptor (firewall rules) and TCP/UDP TPROXY listeners.
type gatewayProvider struct {
    interceptor Interceptor
    cfg         InterceptConfig
    tcpLn       net.Listener
    udpConn     *net.UDPConn
    done        chan struct{}
}

var _ provider.Provider = (*gatewayProvider)(nil)

func newGatewayProvider(interceptor Interceptor, cfg InterceptConfig) *gatewayProvider {
    return &gatewayProvider{
        interceptor: interceptor,
        cfg:         cfg,
        done:        make(chan struct{}),
    }
}

func (p *gatewayProvider) Start(ctx context.Context, handler provider.ConnectionHandler) error {
    port := p.cfg.ListenPort
    if port == 0 {
        port = DefaultListenPort
    }

    // 1. Start TCP TPROXY listener.
    tcpLn, err := newTPROXYTCPListener(ctx, port)
    if err != nil {
        return fmt.Errorf("gateway: tproxy TCP listen: %w", err)
    }
    p.tcpLn = tcpLn

    // 2. Start UDP TPROXY listener.
    udpConn, err := newTPROXYUDPListener(ctx, port)
    if err != nil {
        tcpLn.Close()
        return fmt.Errorf("gateway: tproxy UDP listen: %w", err)
    }
    p.udpConn = udpConn

    // 3. Install intercept rules (after listeners are ready).
    if err := p.interceptor.Install(p.cfg); err != nil {
        tcpLn.Close()
        udpConn.Close()
        return fmt.Errorf("gateway: install intercept: %w", err)
    }

    // 4. TCP accept loop.
    go func() {
        for {
            conn, err := tcpLn.Accept()
            if err != nil {
                select {
                case <-p.done:
                    return
                default:
                    continue
                }
            }
            dest := conn.LocalAddr().String()
            go handler.HandleTCP(conn, dest)
        }
    }()

    // 5. UDP read loop.
    go func() {
        buf := make([]byte, 4096)
        for {
            n, src, dst, err := readUDPOrigDst(p.udpConn, buf)
            if err != nil {
                select {
                case <-p.done:
                    return
                default:
                    continue
                }
            }
            pconn := newTProxyUDPConn(p.udpConn, src, dst, buf[:n])
            go handler.HandleUDP(pconn, dst.String())
        }
    }()

    slog.Info("gateway provider started", "port", port, "interceptor", p.interceptor.Name())
    return nil
}

func (p *gatewayProvider) Close() error {
    select {
    case <-p.done:
        return nil
    default:
        close(p.done)
    }
    // Remove intercept rules first (stops new traffic).
    if p.interceptor != nil {
        p.interceptor.Remove()
    }
    if p.tcpLn != nil {
        p.tcpLn.Close()
    }
    if p.udpConn != nil {
        p.udpConn.Close()
    }
    return nil
}

func (p *gatewayProvider) Name() string { return "gateway" }

        errCh <- g.httpServer.ListenAndServe()
    }()

    select {
    case <-g.ctx.Done():
        g.doDown()
        shutCtx, shutCancel := context.WithTimeout(context.Background(), 5*time.Second)
        defer shutCancel()
        g.httpServer.Shutdown(shutCtx)
        return nil
    case err := <-errCh:
        return err
    }
}

// Stop shuts down the gateway.
func (g *Gateway) Stop() {
    g.cancel()
}

func (g *Gateway) doUp(cfg *GatewayConfig) error {
    g.opMu.Lock()
    defer g.opMu.Unlock()

    // Tear down previous tunnel if running.
    g.closeTunnel()

    // 1. Discover LAN subnets.
    var subnets []string
    var err error
    if len(cfg.LANSubnets) > 0 {
        subnets = cfg.LANSubnets
    } else {
        subnets, err = DiscoverLANSubnets()
        if err != nil {
            return err
        }
    }
    slog.Info("DIAG: gw-lan-discover", "subnets", subnets, "ifaceCount", len(subnets))

    // 2. Resolve exclude IPs.
    excludeIPs, err := resolveExcludeIPs(cfg.Server, cfg.DNS.Direct)
    if err != nil {
        return err
    }

    // 3. Enable IP forwarding.
    var sc sysctlState
    if err := sc.Enable(); err != nil {
        return err
    }

    // 4. Create Interceptor.
    interceptor, err := NewInterceptor()
    if err != nil {
        sc.Restore()
        return err
    }

    // 5. Create gateway provider.
    icfg := InterceptConfig{
        ListenPort: cfg.ListenPort,
        LANSubnets: subnets,
        ExcludeIPs: excludeIPs,
    }
    prov := newGatewayProvider(interceptor, icfg)

    // 6. Set Mode so engine wires DNS middleware (gate: Mode != "proxy").
    cfg.ClientConfig.Mode = "gateway"

    // 7. Create and start engine.
    eng := engine.New()
    eng.SetEventHandler(g)
    ecfg := engineConfig(cfg, prov, g.broadcastSnapshot)
    if err := eng.Start(ecfg); err != nil {
        // Rollback: provider.Close removes intercept rules.
        prov.Close()
        sc.Restore()
        return err
    }

    // 8. Store references under lock.
    g.mu.Lock()
    g.eng = eng
    g.interceptor = interceptor
    g.provider = prov
    g.sysctl = sc
    g.mu.Unlock()

    // 9. Save state for auto-reconnect.
    saveState(&persistedState{Config: cfg, State: "connected", Timestamp: time.Now()})

    return nil
}

func (g *Gateway) doDown() {
    g.opMu.Lock()
    defer g.opMu.Unlock()
    g.closeTunnel()
    saveState(&persistedState{State: "disconnected", Timestamp: time.Now()})
}

// closeTunnel tears down engine + interceptor + sysctl. Caller must hold opMu.
func (g *Gateway) closeTunnel() {
    g.mu.Lock()
    eng := g.eng
    g.eng = nil
    interceptor := g.interceptor
    g.interceptor = nil
    g.provider = nil
    sc := g.sysctl
    g.sysctl = sysctlState{}
    g.mu.Unlock()

    if eng != nil {
        eng.Stop()
    }
    if interceptor != nil {
        interceptor.Remove()
    }
    sc.Restore()
}

func (g *Gateway) OnStatus(status engine.Status) {
    g.mu.Lock()
    g.lastStatus = status
    g.mu.Unlock()
    g.broadcastStatus(status)
}

func (g *Gateway) OnStats(tx, rx uint64) {
    // Stats are delivered via HealthCallback, not OnStats.
}

func (g *Gateway) broadcastSnapshot(snap engine.HealthSnapshot) {
    data, _ := json.Marshal(snap)
    g.broadcastSSE("stats", data)
}

func (g *Gateway) broadcastStatus(status engine.Status) {
    data, _ := json.Marshal(status)

// --- Recovery ---

func (g *Gateway) cleanStaleRules() {
    if i, err := NewInterceptor(); err == nil {
        i.Remove()
    }
}

func (g *Gateway) handlePanic() {
    if r := recover(); r != nil {
        slog.Error("gateway: panic", "err", r, "stack", string(debug.Stack()))
        // Extract-nil-release-close pattern: nil fields under lock, act outside.
        g.mu.Lock()
        interceptor := g.interceptor
        g.interceptor = nil
        sc := g.sysctl
        g.sysctl = sysctlState{}
        g.mu.Unlock()
        if interceptor != nil {
            interceptor.Remove()
        }
        sc.Restore()
    }
}

// tryAutoReconnect loads saved state and reconnects if recently connected.
func (g *Gateway) tryAutoReconnect() {
    st := loadState()
    if st == nil || st.State != "connected" || st.Config == nil {
        return
    }
    // Only auto-reconnect if saved <7 days ago.
    if time.Since(st.Timestamp) > 7*24*time.Hour {
        return
    }
    time.Sleep(3 * time.Second) // brief delay for service startup
    select {
    case <-g.ctx.Done():
        return
    default:
    }
    slog.Info("gateway: auto-reconnecting from saved state")
    if err := g.doUp(st.Config); err != nil {
        slog.Warn("gateway: auto-reconnect failed", "err", err)
    }
}

import (

// Request is the HTTP API request format (matches daemon/).
type Request struct {
    Action string         `json:"action"`
    Params map[string]any `json:"params"`
}

// Response is the HTTP API response format (matches daemon/).
type Response struct {
    Code    int    `json:"code"`
    Message string `json:"message"`
    Data    any    `json:"data,omitempty"`
}

func (g *Gateway) handlePing(w http.ResponseWriter, r *http.Request) {
    writeJSON(w, Response{Code: 0, Message: "pong"})
}

func (g *Gateway) handleCore(w http.ResponseWriter, r *http.Request) {
    if r.Method != http.MethodPost {
        http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
        return
    }

    var req Request
    if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
        writeJSON(w, Response{Code: 1, Message: "bad request"})
        return
    }

    switch req.Action {
    case "up":
        g.handleUp(w, req.Params)
    case "down":
        g.handleDown(w)
    case "status":
        g.handleStatus(w)
    default:
        writeJSON(w, Response{Code: 1, Message: "unknown action"})
    }
}

func (g *Gateway) handleUp(w http.ResponseWriter, params map[string]any) {
    var cfg *GatewayConfig

    if raw, ok := params["config"]; ok {
        data, _ := json.Marshal(raw)
        cfg = &GatewayConfig{}
        if err := json.Unmarshal(data, cfg); err != nil {
            writeJSON(w, Response{Code: 1, Message: "bad config: " + err.Error()})
            return
        }
        SetGatewayDefaults(cfg)
    } else {
        // Try loading from default config path.
        var err error
        cfg, err = LoadGateway(defaultConfigPath())
        if err != nil {
            writeJSON(w, Response{Code: 1, Message: "no config: " + err.Error()})
            return
        }
    }

    go func() {
        if err := g.doUp(cfg); err != nil {
            slog.Warn("gateway: doUp failed", "err", err)
        }
    }()

    writeJSON(w, Response{Code: 0, Message: "connecting"})
}

func (g *Gateway) handleDown(w http.ResponseWriter) {
    go g.doDown()
    writeJSON(w, Response{Code: 0, Message: "disconnecting"})
}

func (g *Gateway) handleStatus(w http.ResponseWriter) {
    g.mu.Lock()
    status := g.lastStatus
    g.mu.Unlock()
    writeJSON(w, Response{Code: 0, Data: status})
}

func (g *Gateway) handleLogLevel(w http.ResponseWriter, r *http.Request) {
    if r.Method != http.MethodPost {
        http.Error(w, "method not allowed", http.StatusMethodNotAllowed)
        return
    }
    var body struct {
        Level string `json:"level"`
    }
    if err := json.NewDecoder(r.Body).Decode(&body); err != nil {
        writeJSON(w, Response{Code: 1, Message: "bad request"})
        return
    }
    config.SetLogLevel(body.Level)
    writeJSON(w, Response{Code: 0, Message: "ok"})
}

func writeJSON(w http.ResponseWriter, v any) {
    w.Header().Set("Content-Type", "application/json")
    json.NewEncoder(w).Encode(v)
}

    return "/etc/k2r/k2r.yml"
}

func (g *Gateway) handleEvents(w http.ResponseWriter, r *http.Request) {
    flusher, ok := w.(http.Flusher)
    if !ok {
        http.Error(w, "streaming not supported", http.StatusInternalServerError)
        return
    }

    w.Header().Set("Content-Type", "text/event-stream")
    w.Header().Set("Cache-Control", "no-cache")
    w.Header().Set("Connection", "keep-alive")
    w.Header().Set("Access-Control-Allow-Origin", "*")

    ch := make(chan []byte, 8)
    g.sseMu.Lock()
    g.sseClients[ch] = struct{}{}
    g.sseMu.Unlock()

    defer func() {
        g.sseMu.Lock()
        delete(g.sseClients, ch)
        g.sseMu.Unlock()
    }()

    // Send current status immediately.
    g.mu.Lock()
    status := g.lastStatus
    g.mu.Unlock()
    g.broadcastStatus(status)

    ctx := r.Context()
    for {
        select {
        case <-ctx.Done():
            return
        case data := <-ch:
            fmt.Fprintf(w, "data: %s\n\n", data)
            flusher.Flush()
        }
    }
}

func (g *Gateway) broadcastSSE(event string, data []byte) {
    msg := []byte(fmt.Sprintf("event: %s\ndata: %s\n\n", event, data))
    g.sseMu.RLock()
    defer g.sseMu.RUnlock()
    for ch := range g.sseClients {
        select {
        case ch <- msg:
        default:
            // Slow client, drop message.
        }
    }
}

// Client is the IPC client used by k2r CLI commands.
type Client struct {
    addr   string
    client *http.Client
}

// NewClient creates a gateway IPC client.
func NewClient(addr string) *Client {
    return &Client{
        addr: addr,
        client: &http.Client{
            Timeout: 30 * time.Second,
        },
    }
}

// Do sends an action to the gateway daemon.
func (c *Client) Do(action string, params map[string]any) (*Response, error) {
    req := Request{Action: action, Params: params}
    body, _ := json.Marshal(req)

    resp, err := c.client.Post(
        fmt.Sprintf("http://%s/api/core", c.addr),
        "application/json",
        bytes.NewReader(body),
    )
    if err != nil {
        return nil, fmt.Errorf("gateway daemon not running (tried %s): %w", c.addr, err)
    }
    defer resp.Body.Close()

    data, _ := io.ReadAll(resp.Body)
    var r Response
    if err := json.Unmarshal(data, &r); err != nil {
        return nil, fmt.Errorf("bad response: %s", data)
    }
    return &r, nil
}

// Ping checks if the daemon is running.
func (c *Client) Ping() error {
    resp, err := c.client.Get(fmt.Sprintf("http://%s/ping", c.addr))
    if err != nil {
        return err
    }
    resp.Body.Close()
    return nil
}

package gateway

import (

const (
    systemdUnit = "k2r.service"
    unitDir     = "/etc/systemd/system"
    initdScript = "/etc/init.d/k2r"
)

// InstallService installs k2r as a system service.
func InstallService() error {
    if isOpenWRT() {
        return installOpenWRT()
    }
    return installSystemd()
}

// UninstallService removes k2r from system services.
func UninstallService() error {
    if isOpenWRT() {
        return uninstallOpenWRT()
    }
    return uninstallSystemd()
}

// IsServiceInstalled returns true if k2r is installed as a system service.
func IsServiceInstalled() bool {
    if isOpenWRT() {
        _, err := os.Stat(initdScript)
        return err == nil
    }
    _, err := os.Stat(filepath.Join(unitDir, systemdUnit))
    return err == nil
}

func isOpenWRT() bool {
    if _, err := os.Stat("/etc/openwrt_release"); err == nil {
        return true
    }
    if _, err := exec.LookPath("fw4"); err == nil {
        return true
    }
    return false
}

// --- systemd ---

func installSystemd() error {
    exe, err := os.Executable()
    if err != nil {
        return err
    }

    unit := fmt.Sprintf(`[Unit]
Description=k2r Gateway
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
ExecStart=%s
Restart=always
RestartSec=5
LimitNOFILE=65535

[Install]
WantedBy=multi-user.target
`, exe)

    path := filepath.Join(unitDir, systemdUnit)
    if err := os.WriteFile(path, []byte(unit), 0644); err != nil {
        return err
    }
    _ = runCmd("systemctl", "daemon-reload")
    _ = runCmd("systemctl", "enable", systemdUnit)
    return runCmd("systemctl", "start", systemdUnit)
}

func uninstallSystemd() error {
    _ = runCmd("systemctl", "stop", systemdUnit)
    _ = runCmd("systemctl", "disable", systemdUnit)
    os.Remove(filepath.Join(unitDir, systemdUnit))
    _ = runCmd("systemctl", "daemon-reload")
    return nil
}

// --- OpenWRT init.d/procd ---

func installOpenWRT() error {
    exe, err := os.Executable()
    if err != nil {
        return err
    }

    script := fmt.Sprintf(`#!/bin/sh /etc/rc.common

START=99
STOP=10
USE_PROCD=1

start_service() {
    procd_open_instance
    procd_set_param command %s
    procd_set_param respawn
    procd_set_param stdout 1
    procd_set_param stderr 1
    procd_close_instance
}
`, exe)

    if err := os.WriteFile(initdScript, []byte(script), 0755); err != nil {
        return err
    }
    _ = runCmd(initdScript, "enable")
    return runCmd(initdScript, "start")
}

func uninstallOpenWRT() error {
    _ = runCmd(initdScript, "stop")
    _ = runCmd(initdScript, "disable")
    os.Remove(initdScript)
    return nil
}

import (

var (

    commit  = "unknown"

func main() {
    fs := flag.NewFlagSet("k2r", flag.ContinueOnError)
    configPath := fs.String("c", "", "config file path")
    showVersion := fs.Bool("v", false, "show version")
    if err := fs.Parse(os.Args[1:]); err != nil {

    case "setup":

    case "ctl":
        cmdCtl(args[1:])
    case "service":
        cmdService(args[1:])
    default:
        fmt.Fprintf(os.Stderr, "unknown command: %s\n", args[0])
        fmt.Fprintln(os.Stderr, "usage: k2r [setup|ctl|service] or k2r -c config.yml")
        os.Exit(1)
    }
}

func runDaemon(configPath string) {
    // Set up logging.
    if configPath == "" {
        configPath = "/etc/k2r/k2r.yml"
    }
    cfg, err := gateway.LoadGateway(configPath)
    if err != nil {
        // Start without config — will wait for API up command.
        cfg = &gateway.GatewayConfig{}
        gateway.SetGatewayDefaults(cfg)
    }
    closer, err := config.SetupLogging(cfg.Log, "k2r")
    if err != nil {
        fmt.Fprintf(os.Stderr, "logging: %v\n", err)
        os.Exit(1)
    }
    defer closer()

    g := gateway.New()
    if err := g.Run(cfg.Listen); err != nil {
        fmt.Fprintf(os.Stderr, "gateway: %v\n", err)
        os.Exit(1)
    }
}

func cmdSetup(args []string) {
    fs := flag.NewFlagSet("setup", flag.ExitOnError)
    configFile := fs.String("c", "", "existing config file to install")
    fs.Parse(args)

    var cfg *gateway.GatewayConfig
    if *configFile != "" {
        var err error
        cfg, err = gateway.LoadGateway(*configFile)
        if err != nil {
            fmt.Fprintf(os.Stderr, "load config: %v\n", err)
            os.Exit(1)
        }
    } else if fs.NArg() > 0 {
        cfg = gateway.GatewayFromURL(fs.Arg(0))
    } else {
        fmt.Fprintln(os.Stderr, "usage: k2r setup <k2v5://URL> or k2r setup -c config.yml")
        os.Exit(1)
    }

    // Save config.
    os.MkdirAll("/etc/k2r", 0755)
    data, _ := json.MarshalIndent(cfg, "", "  ")
    if err := os.WriteFile("/etc/k2r/k2r.yml", data, 0600); err != nil {
        fmt.Fprintf(os.Stderr, "save config: %v\n", err)
        os.Exit(1)
    }

    // Install and start service.
    if err := gateway.InstallService(); err != nil {
        fmt.Fprintf(os.Stderr, "install service: %v\n", err)
        os.Exit(1)
    }

    fmt.Println("k2r installed and started.")
    fmt.Println("  k2r ctl status   — check connection")
    fmt.Println("  k2r ctl down     — disconnect")
}

func cmdCtl(args []string) {
    if len(args) == 0 {
        fmt.Fprintln(os.Stderr, "usage: k2r ctl [up|down|status]")
        os.Exit(1)
    }

    client := gateway.NewClient(gateway.DefaultGatewayAddr)

    switch args[0] {
    case "up":
        params := map[string]any{}
        if len(args) > 1 {
            // URL or config path.
            arg := args[1]
            if _, err := os.Stat(arg); err == nil {
                cfg, err := gateway.LoadGateway(arg)
                if err != nil {
                    fmt.Fprintf(os.Stderr, "load config: %v\n", err)
                    os.Exit(1)
                }
                params["config"] = cfg
            } else {
                cfg := gateway.GatewayFromURL(arg)
                params["config"] = cfg
            }
        }
        resp, err := client.Do("up", params)
        if err != nil {
            fmt.Fprintf(os.Stderr, "%v\n", err)
            os.Exit(1)
        }
        fmt.Println(resp.Message)

    case "down":
        resp, err := client.Do("down", nil)
        if err != nil {
            fmt.Fprintf(os.Stderr, "%v\n", err)
            os.Exit(1)
        }
        fmt.Println(resp.Message)

    case "status":
        resp, err := client.Do("status", nil)
        if err != nil {
            fmt.Fprintf(os.Stderr, "%v\n", err)
            os.Exit(1)
        }
        data, _ := json.MarshalIndent(resp.Data, "", "  ")
        fmt.Println(string(data))

    default:
        fmt.Fprintf(os.Stderr, "unknown ctl command: %s\n", args[0])
        os.Exit(1)
    }
}

    if len(args) == 0 {
        fmt.Fprintln(os.Stderr, "usage: k2r service [install|uninstall]")
        os.Exit(1)
