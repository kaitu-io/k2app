> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Deliver k2p (peer-to-peer home access) end-to-end: URL parsing, signal server, peer auth, stub relay, Home/Away integration, CLI commands, and e2e test.

**Architecture:** Home runs `server.ProxyHandler` behind NAT. Away connects via hole punch (direct) or relay (fallback). Data plane is k2v5 (QUIC + k2cc). k2s acts as zero-trust signal/relay server. **Tech Stack:** Go 1.25, QUIC (apernet/quic-go), gorilla/websocket, pion/stun v3, deadlock mutex, slog diagnostics, `golang.org/x/crypto/bcrypt`

```go
// ParseURL parses a k2p:// URL into Config.
// Format: k2p://UDID:SECRET@UDID.signal-domain:port?pin=...&ip=...&ech=...&fp=...
func ParseURL(raw string) (Config, error) {
	u, err := url.Parse(raw)
	if err != nil {
		return Config{}, fmt.Errorf("k2p: parse URL: %w", err)
	}
	if u.Scheme != "k2p" {
		return Config{}, fmt.Errorf("k2p: unsupported scheme %q, want k2p", u.Scheme)
	}
	if u.User == nil {
		return Config{}, fmt.Errorf("k2p: missing auth (udid:secret@)")
	}

	udid := u.User.Username()
	secret, _ := u.User.Password()
	if err := ValidateUDID(udid); err != nil {
		return Config{}, err
	}
	if secret == "" {
		return Config{}, fmt.Errorf("k2p: missing peer secret in URL")
	}

	host := u.Hostname()
	port := u.Port()
	if port == "" {
		port = "443"
	}

	// Strip UDID prefix from host: "home-abc.proxy.kaitu.io" → "proxy.kaitu.io"
	baseDomain := host
	if idx := strings.Index(host, "."); idx > 0 {
		baseDomain = host[idx+1:]
	}
	signalServer := "https://" + baseDomain + ":" + port

	q := u.Query()
	cfg := Config{
		HomeUDID:     udid,
		PeerSecret:   secret,
		SignalServer: signalServer,
		Pin:          q.Get("pin"),
	}

	// Populate WireCfg from query params (same as k2v5).
	cfg.WireCfg = wire.Config{
		Host:        baseDomain,
		Fingerprint: "chrome",
	}
	if v := q.Get("ip"); v != "" {
		cfg.WireCfg.IPv4 = v
	}
	if v := q.Get("ip6"); v != "" {
		cfg.WireCfg.IPv6 = v
	}
	if v := q.Get("ech"); v != "" {
		cfg.WireCfg.ECH = v
	}
	if v := q.Get("fp"); v != "" {
		cfg.WireCfg.Fingerprint = v
	}

	return cfg, nil
}
```

```go
// BuildURL constructs a k2p:// URL from components.
func BuildURL(udid, secret, signalDomain string, port int, pin string, extra url.Values) string {
	u := &url.URL{
		Scheme: "k2p",
		User:   url.UserPassword(udid, secret),
		Host:   fmt.Sprintf("%s.%s:%d", udid, signalDomain, port),
	}
	q := make(url.Values)
	if pin != "" {
		q.Set("pin", pin)
	}
	for k, vs := range extra {
		for _, v := range vs {
			q.Set(k, v)
		}
	}
	u.RawQuery = q.Encode()
	return u.String()
}
```

```go
// k2p/auth.go
package k2p

import (
	"encoding/binary"
	"fmt"
	"io"

	"golang.org/x/crypto/bcrypt"
)

const authVersion = 0x01

// WriteAuthFrame writes the peer auth frame: [version(1)][len(2 BE)][secret(N)].
func WriteAuthFrame(w io.Writer, secret string) error {
	secretBytes := []byte(secret)
	if len(secretBytes) > 65535 {
		return fmt.Errorf("k2p: secret too long (%d bytes)", len(secretBytes))
	}
	header := make([]byte, 3)
	header[0] = authVersion
	binary.BigEndian.PutUint16(header[1:3], uint16(len(secretBytes)))
	if _, err := w.Write(header); err != nil {
		return fmt.Errorf("k2p: write auth header: %w", err)
	}
	if _, err := w.Write(secretBytes); err != nil {
		return fmt.Errorf("k2p: write auth secret: %w", err)
	}
	return nil
}

// ReadAuthFrame reads a peer auth frame and returns the secret.
func ReadAuthFrame(r io.Reader) (string, error) {
	header := make([]byte, 3)
	if _, err := io.ReadFull(r, header); err != nil {
		return "", fmt.Errorf("k2p: read auth header: %w", err)
	}
	if header[0] != authVersion {
		return "", fmt.Errorf("k2p: unsupported auth version 0x%02x", header[0])
	}
	length := binary.BigEndian.Uint16(header[1:3])
	secret := make([]byte, length)
	if _, err := io.ReadFull(r, secret); err != nil {
		return "", fmt.Errorf("k2p: read auth secret: %w", err)
	}
	return string(secret), nil
}

// Auth response codes.
const (
	AuthOK       byte = 0x00
	AuthRejected byte = 0x01
)

// HashSecret returns a bcrypt hash of the secret for persistent storage.
func HashSecret(secret string) (string, error) {
	hash, err := bcrypt.GenerateFromPassword([]byte(secret), bcrypt.DefaultCost)
	if err != nil {
		return "", fmt.Errorf("k2p: hash secret: %w", err)
	}
	return string(hash), nil
}

// VerifySecret checks a plaintext secret against a bcrypt hash.
func VerifySecret(secret, hash string) bool {
	return bcrypt.CompareHashAndPassword([]byte(hash), []byte(secret)) == nil
}
```

```go
// signal/store.go
package signal

import (
	"crypto/rand"
	"encoding/hex"
	"fmt"
	"time"

	"github.com/sasha-s/go-deadlock"
	"golang.org/x/crypto/bcrypt"
)

// Store is the in-memory peer registry.
type Store struct {
	mu    deadlock.RWMutex
	peers map[string]*PeerEntry
}

// NewStore creates an empty peer store.
func NewStore() *Store {
	return &Store{peers: make(map[string]*PeerEntry)}
}

// Claim registers a UDID for the first time and returns a registration token.
// Returns error if UDID is already claimed.
// NOTE: bcrypt is called outside the lock per CLAUDE.md concurrency rules.
func (s *Store) Claim(udid string) (string, error) {
	// Generate token and hash OUTSIDE the lock (bcrypt is ~300ms).
	token := generateToken()
	hash, err := bcrypt.GenerateFromPassword([]byte(token), bcrypt.DefaultCost)
	if err != nil {
		return "", fmt.Errorf("signal: hash token: %w", err)
	}

	s.mu.Lock()
	defer s.mu.Unlock()

	if _, exists := s.peers[udid]; exists {
		return "", fmt.Errorf("signal: UDID %q already claimed", udid)
	}

	s.peers[udid] = &PeerEntry{
		UDID:         udid,
		RegTokenHash: string(hash),
		UpdatedAt:    time.Now(),
	}
	return token, nil
}

// Update updates a peer's registration. Token must match the stored hash.
// NOTE: bcrypt verify is called outside the lock per CLAUDE.md concurrency rules.
func (s *Store) Update(udid, token string, req RegisterRequest) error {
	// Copy hash under lock, release, verify outside.
	s.mu.RLock()
	entry, ok := s.peers[udid]
	if !ok {
		s.mu.RUnlock()
		return fmt.Errorf("signal: UDID %q not found", udid)
	}
	storedHash := entry.RegTokenHash
	s.mu.RUnlock()

	if err := bcrypt.CompareHashAndPassword([]byte(storedHash), []byte(token)); err != nil {
		return fmt.Errorf("signal: invalid registration token")
	}

	s.mu.Lock()
	defer s.mu.Unlock()
	entry, ok = s.peers[udid]
	if !ok {
		return fmt.Errorf("signal: UDID %q not found", udid)
	}
	entry.Addr = req.Addr
	entry.NATType = req.NATType
	entry.Networks = req.Networks

		Addr:      entry.Addr,

		DNS:       entry.DNS,
		UpdatedAt: entry.UpdatedAt,
	}, true

	defer s.mu.Unlock()

		entry.WS = ws
	}

	defer s.mu.RUnlock()

		return entry.WS
	}
	return nil
```

```go
func (h *Handler) handleClaim(w http.ResponseWriter, r *http.Request) {

		return

		if strings.Contains(err.Error(), "invalid registration token") {

			return

		http.Error(w, "not found", 404)
		return

		UDID:     entry.UDID,
		Addr:     entry.Addr,
		NATType:  entry.NATType,
		Networks: entry.Networks,
		DNS:      entry.DNS,
		Online:   h.store.IsOnline(udid),
	})

		return

		http.Error(w, "home not connected", 404)
		return

	if err := ws.WriteMessage(websocket.TextMessage, msg); err != nil {
		slog.Warn("signal: ws write to home failed", "udid", req.HomeUDID, "err", err)
		http.Error(w, "home unreachable", 502)
		return
	}
	w.WriteHeader(200)
}

func (h *Handler) handleWS(w http.ResponseWriter, r *http.Request) {
	udid := r.URL.Query().Get("udid")
	token := r.URL.Query().Get("token")
	if udid == "" || token == "" {
		http.Error(w, "missing udid or token", 400)
		return
	}

	// Validate token against store.
	entry, ok := h.store.Get(udid)
	if !ok {
		http.Error(w, "not found", 404)
		return
	}
	_ = entry // Token validation happens in store — but Get doesn't expose hash.
	// For WS auth, we re-validate via Update with a no-op.
	// Simpler: store validates token directly.
	if !h.store.ValidateToken(udid, token) {
		http.Error(w, "invalid token", 403)
		return
	}

	conn, err := h.upgrader.Upgrade(w, r, nil)
	if err != nil {
		return
	}
	h.store.SetWS(udid, conn)

	// Hold connection open; clean up on disconnect.
	defer func() {
		h.store.SetWS(udid, nil)
		conn.Close()
	}()

	// Read loop (keep connection alive, handle pong).
	for {
		if _, _, err := conn.ReadMessage(); err != nil {
			return
		}
	}
}

	auth := r.Header.Get("Authorization")
	if strings.HasPrefix(auth, "Bearer ") {
		return auth[7:]
	}
	return ""
```

```go
// signal/relay.go
package signal

import (
	"log/slog"
	"net/http"

	"github.com/gorilla/websocket"
	"github.com/sasha-s/go-deadlock"
)

// relayPair holds home and away WS connections for a UDID relay session.
type relayPair struct {
	home *websocket.Conn
	away *websocket.Conn
}

// Relay manages WS binary relay sessions between Home and Away.
type Relay struct {
	mu       deadlock.Mutex
	pairs    map[string]*relayPair
	upgrader websocket.Upgrader
}

// NewRelay creates a relay manager.
func NewRelay() *Relay {
	return &Relay{
		pairs: make(map[string]*relayPair),
		upgrader: websocket.Upgrader{CheckOrigin: func(r *http.Request) bool { return true }},
	}
}

// HandleRelay handles GET /k2p/relay?udid=...&peer=<peer-udid>.
// The "peer" param identifies the connecting party. Home connects with peer=<home-udid>,
// Away connects with peer=<away-udid>. The relay matches by UDID.
// For simplicity, the first connection is treated as "home" and the second as "away".
func (rl *Relay) HandleRelay(w http.ResponseWriter, r *http.Request) {
	udid := r.URL.Query().Get("udid")
	role := r.URL.Query().Get("role") // "home" or "away"
	if udid == "" || (role != "home" && role != "away") {
		http.Error(w, "missing udid or invalid role", 400)
		return
	}

	conn, err := rl.upgrader.Upgrade(w, r, nil)
	if err != nil {
		return
	}

	rl.mu.Lock()
	pair, ok := rl.pairs[udid]
	if !ok {
		pair = &relayPair{}
		rl.pairs[udid] = pair
	}
	var peer *websocket.Conn
	if role == "home" {
		pair.home = conn
		peer = pair.away
	} else {
		pair.away = conn
		peer = pair.home
	}
	rl.mu.Unlock()

	// If peer is already connected, start forwarding in both directions.
	if peer != nil {
		rl.forward(udid, conn, peer, role)
	} else {
		// Wait for peer by reading (blocks until peer joins or disconnect).
		rl.waitAndForward(udid, conn, role)
	}
}

func (rl *Relay) forward(udid string, src, dst *websocket.Conn, srcRole string) {
	defer rl.cleanup(udid)
	for {
		msgType, msg, err := src.ReadMessage()
		if err != nil {
			return
		}
		if msgType != websocket.BinaryMessage {
			continue // Only forward binary messages.
		}
		if err := dst.WriteMessage(websocket.BinaryMessage, msg); err != nil {
			return
		}
	}
}

func (rl *Relay) waitAndForward(udid string, conn *websocket.Conn, role string) {
	defer rl.cleanup(udid)
	// Poll for peer arrival.
	for {
		rl.mu.Lock()
		pair, ok := rl.pairs[udid]
		if !ok {
			rl.mu.Unlock()
			return
		}
		var peer *websocket.Conn
		if role == "home" {
			peer = pair.away
		} else {
			peer = pair.home
		}
		rl.mu.Unlock()

		if peer != nil {
			rl.forward(udid, conn, peer, role)
			return
		}

		// Brief block on read to detect disconnect.
		conn.SetReadDeadline(deadlineNow100ms())
		_, _, err := conn.ReadMessage()
		if err != nil {
			if websocket.IsCloseError(err, websocket.CloseNormalClosure, websocket.CloseGoingAway) {
				return
			}
			if isTimeout(err) {
				continue
			}
			return
		}
	}
}

func (rl *Relay) cleanup(udid string) {
	rl.mu.Lock()
	pair, ok := rl.pairs[udid]
	if ok {
		if pair.home != nil {
			pair.home.Close()
		}
		if pair.away != nil {
			pair.away.Close()
		}
		delete(rl.pairs, udid)
	}
	rl.mu.Unlock()
	slog.Debug("signal: relay session cleaned up", "udid", udid)
}
```

```go
// k2p/relay.go
package k2p

import (
	"fmt"
	"net"
	"time"

	"github.com/gorilla/websocket"
	"github.com/sasha-s/go-deadlock"
)

// RelayConn wraps a WebSocket connection to implement net.PacketConn.
// Each WS binary message = one UDP datagram.
type RelayConn struct {
	ws        *websocket.Conn
	peerAddr  net.Addr // synthetic peer address
	readDeadline time.Time
	mu        deadlock.Mutex // protects ws.WriteMessage (per CLAUDE.md: never sync.Mutex)
}

// DialRelay opens a WebSocket relay connection and returns it as a net.PacketConn.
func DialRelay(wsURL string) (*RelayConn, error) {
	conn, _, err := websocket.DefaultDialer.Dial(wsURL, nil)
	if err != nil {
		return nil, fmt.Errorf("k2p: relay dial: %w", err)
	}
	return &RelayConn{
		ws:       conn,
		peerAddr: &net.UDPAddr{IP: net.IPv4(0, 0, 0, 0), Port: 0},
	}, nil
}

// ReadFrom reads the next WS binary message as a datagram.
func (rc *RelayConn) ReadFrom(p []byte) (n int, addr net.Addr, err error) {
	if !rc.readDeadline.IsZero() {
		rc.ws.SetReadDeadline(rc.readDeadline)
	}
	_, msg, err := rc.ws.ReadMessage()
	if err != nil {
		return 0, nil, err
	}
	n = copy(p, msg)
	return n, rc.peerAddr, nil
}

// WriteTo sends p as a WS binary message. addr is ignored (relay knows the peer).
func (rc *RelayConn) WriteTo(p []byte, _ net.Addr) (n int, err error) {
	rc.mu.Lock()
	defer rc.mu.Unlock()
	if err := rc.ws.WriteMessage(websocket.BinaryMessage, p); err != nil {
		return 0, err
	}
	return len(p), nil
}

func (rc *RelayConn) Close() error                       { return rc.ws.Close() }
func (rc *RelayConn) LocalAddr() net.Addr                { return &net.UDPAddr{} }
func (rc *RelayConn) SetDeadline(t time.Time) error      { rc.readDeadline = t; return nil }
func (rc *RelayConn) SetReadDeadline(t time.Time) error  { rc.readDeadline = t; return nil }
func (rc *RelayConn) SetWriteDeadline(t time.Time) error { return nil }
```

```go
// k2p/networks.go
package k2p

import "net"

// DetectPrivateNetworks returns the host's private LAN CIDRs and a DNS guess
// (first IP in the first private subnet, typically the gateway).
func DetectPrivateNetworks() (cidrs []string, dns string) {
	ifaces, err := net.Interfaces()
	if err != nil {
		return nil, ""
	}
	for _, iface := range ifaces {
		if iface.Flags&net.FlagUp == 0 || iface.Flags&net.FlagLoopback != 0 {
			continue
		}
		addrs, err := iface.Addrs()
		if err != nil {
			continue
		}
		for _, addr := range addrs {
			ipnet, ok := addr.(*net.IPNet)
			if !ok || !ipnet.IP.IsPrivate() {
				continue
			}
			// Skip IPv6 link-local.
			if ipnet.IP.To4() == nil {
				continue
			}
			cidrs = append(cidrs, ipnet.String())
			if dns == "" {
				gw := make(net.IP, len(ipnet.IP.To4()))
				copy(gw, ipnet.IP.To4())
				gw[3] = 1
				dns = gw.String()
			}
		}
	}
	return cidrs, dns
}
```

- [ ] **Step 2: Update handleAwayConnection to include peer auth**

Peer auth happens OUTSIDE `handleAwayConnection`, in the consumer of `<-chan wire.Listener`.

The approach: `handleAwayConnection` returns `wire.Listener` as before (no changes to the QUIC accept flow). The consumer (`peer_run.go` or the caller) handles peer auth on the first TCP connection from `ln.AcceptTCP()`. Specifically, after the QUIC connection is established, Away opens stream 0 for auth. On the Home side, `QUICServer.AcceptTCP` accepts that first stream. The caller (peer_run.go loop) checks if it's an auth stream:

**Alternative (simpler) approach**: Peer auth is done by Away opening stream 0 with the auth frame BEFORE any proxy streams. Home wraps the `wire.Listener` with a `PeerAuthListener` that intercepts the first `AcceptTCP` call, reads the auth frame, and verifies. If auth fails, it closes the connection and returns error. If auth succeeds, subsequent `AcceptTCP` calls pass through normally.

```go
// k2p/auth.go — add PeerAuthListener wrapper

// PeerAuthListener wraps a wire.Listener and verifies peer auth on the first stream.
type PeerAuthListener struct {
	inner      wire.Listener
	secretHash string
	authDone   bool
	mu         deadlock.Mutex
}

// WrapWithAuth wraps a wire.Listener with peer authentication.
func WrapWithAuth(ln wire.Listener, secretHash string) *PeerAuthListener {
	return &PeerAuthListener{inner: ln, secretHash: secretHash}
}

func (p *PeerAuthListener) AcceptTCP(ctx context.Context) (net.Conn, string, error) {
	conn, addr, err := p.inner.AcceptTCP(ctx)
	if err != nil {
		return nil, "", err
	}

	p.mu.Lock()
	needAuth := !p.authDone
	p.mu.Unlock()

	if needAuth {
		// First stream is the auth stream.
		secret, err := ReadAuthFrame(conn)
		if err != nil {
			conn.Close()
			return nil, "", fmt.Errorf("k2p: read auth: %w", err)
		}
		if !VerifySecret(secret, p.secretHash) {
			conn.Write([]byte{AuthRejected})
			conn.Close()
			return nil, "", fmt.Errorf("k2p: peer auth rejected")
		}
		conn.Write([]byte{AuthOK})
		conn.Close() // Close the auth stream.

		p.mu.Lock()
		p.authDone = true
		p.mu.Unlock()

		// Accept the next stream (actual proxy traffic).
		return p.inner.AcceptTCP(ctx)
	}

	return conn, addr, nil
}

func (p *PeerAuthListener) ServeUDP(ctx context.Context, handler wire.UDPHandler) error {
	return p.inner.ServeUDP(ctx, handler)
}

func (p *PeerAuthListener) Close() error {
	return p.inner.Close()
}
```

Then in `handleAwayConnection`, after creating the listener, wrap it:


This keeps `handleAwayConnection`'s return type as `wire.Listener` (since `PeerAuthListener` implements it) and doesn't consume the QUIC accept prematurely.

```go
func buildPinTLSConfig(cfg Config) *tls.Config {
	tlsCfg := &tls.Config{
		NextProtos:         []string{"h3"},
		InsecureSkipVerify: true, // skip CA chain (self-signed)
	}
	if cfg.Pin != "" {
		expectedPins, err := wire.ParsePins(cfg.Pin)
		if err == nil && len(expectedPins) > 0 {
			tlsCfg.VerifyPeerCertificate = func(rawCerts [][]byte, _ [][]*x509.Certificate) error {
				return verifyPins(rawCerts, expectedPins)
			}
		}
	}
	return tlsCfg
}

func verifyPins(rawCerts [][]byte, expected [][32]byte) error {
	for _, raw := range rawCerts {
		cert, err := x509.ParseCertificate(raw)
		if err != nil {
			continue
		}
		pubDER, err := x509.MarshalPKIXPublicKey(cert.PublicKey)
		if err != nil {
			continue
		}
		hash := sha256.Sum256(pubDER)
		for _, exp := range expected {
			if hash == exp {
				return nil
			}
		}
	}
	return fmt.Errorf("k2p: no certificate matched pin")
}
```

```go
	tr := &quic.Transport{Conn: relay}
	tlsCfg := buildPinTLSConfig(cfg)
	quicCfg := wire.DefaultQUICConfig()

	// to WS, but quic-go needs a non-zero destination to function.

	}

	// Peer auth.
	authStream, err := conn.OpenStreamSync(ctx)
	if err != nil {
		conn.CloseWithError(1, "auth failed")
		tr.Close()
		return nil, err
	}
	if err := WriteAuthFrame(authStream, cfg.PeerSecret); err != nil {
```

```go
// test/k2p_e2e_test.go
package test

import (
	"context"
	"net"
	"net/http/httptest"
	"testing"
	"time"

	"github.com/kaitu-io/k2/k2p"
	"github.com/kaitu-io/k2/signal"
	"github.com/kaitu-io/k2/wire"
)

func TestK2P_E2E_DirectPunch(t *testing.T) {
	if testing.Short() {
		t.Skip("e2e test")
	}

	// 1. Start signal server.
	sigHandler := signal.NewHandler()
	sigSrv := httptest.NewServer(sigHandler)
	defer sigSrv.Close()

	// 2. Generate Home cert + secret.
	certPair, _ := wire.GenerateCertPair()
	secret := "test-peer-secret"
	secretHash, _ := k2p.HashSecret(secret)
	pinStr := certPair.FormatPins()

	// 3. Home claims UDID.
	homeSignal := k2p.NewSignalClient(sigSrv.URL, "")
	regToken, err := homeSignal.Claim(context.Background(), "test-home")
	if err != nil {
		t.Fatal(err)
	}

	// 4. Start mock STUN servers.
	_, stunAddr1 := startTestSTUNServer(t)
	_, stunAddr2 := startTestSTUNServer(t)

	// 5. Home listens.
	ctx, cancel := context.WithTimeout(context.Background(), 15*time.Second)
	defer cancel()

	homeCfg := k2p.Config{
		HomeUDID:     "test-home",
		SignalServer: sigSrv.URL,
		RegToken:     regToken,
		SecretHash:   secretHash,
		TLSCert:      &certPair.EC,
		STUNServers:  []string{stunAddr1, stunAddr2},
	}
	listenerCh, err := k2p.Listen(ctx, homeCfg)
	if err != nil {
		t.Fatal(err)
	}

	// 6. Away dials.
	awayCfg := k2p.Config{
		HomeUDID:     "test-home",
		SignalServer: sigSrv.URL,
		PeerSecret:   secret,
		Pin:          pinStr,
		STUNServers:  []string{stunAddr1, stunAddr2},
	}
	result, err := k2p.Dial(ctx, awayCfg)
	if err != nil {
		t.Fatalf("Dial: %v", err)
	}
	if result.TransportType != "p2p-direct" {
		t.Errorf("transport = %q, want p2p-direct", result.TransportType)
	}
	defer result.Dialer.Close()

	// 7. Verify Home received the connection.
	select {
	case ln := <-listenerCh:
		if ln == nil {
			t.Fatal("nil listener")
		}
		ln.Close()
	case <-time.After(5 * time.Second):
		t.Fatal("Home did not receive connection")
	}
}

func TestK2P_E2E_AuthRejection(t *testing.T) {
	if testing.Short() {
		t.Skip("e2e test")
	}

	sigHandler := signal.NewHandler()
	sigSrv := httptest.NewServer(sigHandler)
	defer sigSrv.Close()

	certPair, _ := wire.GenerateCertPair()
	secretHash, _ := k2p.HashSecret("correct-secret")

	homeSignal := k2p.NewSignalClient(sigSrv.URL, "")
	regToken, _ := homeSignal.Claim(context.Background(), "test-home")

	_, stunAddr1 := startTestSTUNServer(t)
	_, stunAddr2 := startTestSTUNServer(t)

	ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
	defer cancel()

	homeCfg := k2p.Config{
		HomeUDID:     "test-home",
		SignalServer: sigSrv.URL,
		RegToken:     regToken,
		SecretHash:   secretHash,
		TLSCert:      &certPair.EC,
		STUNServers:  []string{stunAddr1, stunAddr2},
	}
	k2p.Listen(ctx, homeCfg)

	// Away tries with WRONG secret.
	awayCfg := k2p.Config{
		HomeUDID:     "test-home",
		SignalServer: sigSrv.URL,
		PeerSecret:   "wrong-secret",
		Pin:          certPair.FormatPins(),
		STUNServers:  []string{stunAddr1, stunAddr2},
	}
	_, err := k2p.Dial(ctx, awayCfg)
	if err == nil {
		t.Fatal("expected auth rejection error")
	}
	t.Logf("correctly rejected: %v", err)
}
```
