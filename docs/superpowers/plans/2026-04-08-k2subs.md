# k2subs:// Subscription URL Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add `k2subs://` subscription URL support — a URL scheme that fetches a weighted list of k2v5:// tunnel servers, enabling dynamic server selection and failover.

**Architecture:** New `config/subscription.go` module handles URL parsing, HTTP fetch, disk caching, weighted random selection, and background refresh. Daemon resolves k2subs:// URLs to k2v5:// before calling engine.Start() — engine sees only k2v5:// and requires zero changes. CLI is updated to accept k2subs:// URLs. `k2 connect` command is deleted (functionality covered by `k2 up`).

**Tech Stack:** Go stdlib (`net/http`, `net/url`, `encoding/json`, `crypto/sha256`, `math/rand/v2`), `go-deadlock` for mutex.

**Spec:** `docs/superpowers/specs/2026-04-08-k2subs-design.md`

**Safety invariants (must hold throughout implementation):**

1. **Lock safety**: `Subscription.mu` is a state lock (per CLAUDE.md rules). No I/O, no external calls, no Close(), no waiting under lock. HTTP fetch happens entirely outside lock. `Subscription.mu` is standalone — no nesting with engine/daemon locks.
2. **Persisted state**: `saveState()` must persist the **original** k2subs:// URL, never the resolved k2v5:// URL. Otherwise auto-reconnect after crash connects to one fixed server instead of re-resolving from the subscription list.
3. **Crash-safe disk cache**: Use write-to-temp + atomic rename to prevent partial cache files from corrupting state on crash during write.
4. **Goroutine lifecycle**: `refreshLoop` goroutine is bounded by context cancellation + 30s fetch timeout. `Close()` waits for goroutine exit via done channel. No goroutine leak possible.
5. **Partial resolution cleanup**: If `resolveSubscriptions` fails on the Nth route, Close() all already-created Subscription objects to prevent goroutine/resource leaks.
6. **Response body limit**: Limit HTTP response body to 1MB to prevent OOM from a malicious/misconfigured server.

```go
// config/subscription.go
package config

import (
	"context"
	"crypto/sha256"
	"encoding/json"
	"errors"
	"fmt"
	"io"
	"math/rand/v2"
	"net/http"
	"net/url"
	"os"
	"path/filepath"
	"strconv"
	"strings"
	"time"

	"github.com/sasha-s/go-deadlock"
)

// Tunnel represents a single tunnel server returned by a subscription endpoint.
type Tunnel struct {
	URL    string `json:"url"`
	Weight int    `json:"weight"`
}

// subsResponse is the JSON response from the subscription endpoint.
type subsResponse struct {
	Tunnels []Tunnel `json:"tunnels"`
	Refresh *int     `json:"refresh,omitempty"` // server-controlled refresh interval (seconds)
}

// Subscription manages a k2subs:// subscription: fetching tunnel lists,
// caching results, and weighted random server selection.
// Independent of engine/wire/daemon — only depends on stdlib + deadlock.
type Subscription struct {
	// Immutable after construction.
	raw       string
	endpoint  string        // https:// URL
	creds     string        // "user:pass"
	refresh   time.Duration // client-side fallback refresh interval
	cacheFile string        // disk cache path ("" = no disk cache)

	// State (mu-protected).
	mu        deadlock.Mutex
	tunnels   []Tunnel
	fetchedAt time.Time
	interval  time.Duration // effective refresh interval (from last fetch or client default)

	cancel context.CancelFunc
	done   chan struct{}
}

// NewSubscription parses a k2subs:// URL and creates a Subscription.
// If cacheDir is non-empty, loads any existing disk cache.
func NewSubscription(raw string, cacheDir string) (*Subscription, error) {
	if raw == "" {
		return nil, errors.New("subscription: empty URL")
	}

	u, err := url.Parse(raw)
	if err != nil {
		return nil, fmt.Errorf("subscription: parse URL: %w", err)
	}
	if u.Scheme != "k2subs" {
		return nil, fmt.Errorf("subscription: unsupported scheme %q, want k2subs", u.Scheme)
	}
	if u.User == nil {
		return nil, errors.New("subscription: missing credentials (user:pass@)")
	}
	user := u.User.Username()
	pass, hasPass := u.User.Password()
	if user == "" || !hasPass || pass == "" {
		return nil, errors.New("subscription: missing user or password in credentials")
	}

	// Build HTTPS endpoint: replace scheme, strip userinfo and query.
	endpoint := &url.URL{
		Scheme: "https",
		Host:   u.Host,
		Path:   u.Path,
	}

	// Parse refresh parameter.
	refresh := 1800 * time.Second // default 30 min
	if v := u.Query().Get("refresh"); v != "" {
		secs, err := strconv.Atoi(v)
		if err != nil {
			return nil, fmt.Errorf("subscription: invalid refresh %q: %w", v, err)
		}
		refresh = time.Duration(secs) * time.Second
	}

	s := &Subscription{
		raw:      raw,
		endpoint: endpoint.String(),
		creds:    user + ":" + pass,
		refresh:  refresh,
		interval: refresh,
	}

	// Disk cache.
	if cacheDir != "" {
		hash := sha256.Sum256([]byte(raw))
		s.cacheFile = filepath.Join(cacheDir, fmt.Sprintf("subs-%x.json", hash[:8]))
		s.loadCache()
	}

	return s, nil
}
```

```go
// Fetch retrieves the tunnel list from the subscription endpoint.
// httpClient is provided by the caller (controls direct vs proxy routing).
// On success, updates in-memory list + disk cache. On failure, preserves old data.
func (s *Subscription) Fetch(ctx context.Context, httpClient *http.Client) error {
	req, err := http.NewRequestWithContext(ctx, "GET", s.endpoint, nil)
	if err != nil {
		return fmt.Errorf("subscription: build request: %w", err)
	}

	// Split creds into user:pass for Basic Auth.
	user, pass := splitCreds(s.creds)
	req.SetBasicAuth(user, pass)

	resp, err := httpClient.Do(req)
	if err != nil {
		return fmt.Errorf("subscription: fetch: %w", err)
	}
	defer resp.Body.Close()

	if resp.StatusCode != 200 {
		return fmt.Errorf("subscription: server returned %d", resp.StatusCode)
	}

	// Limit response body to 1MB to prevent OOM from malicious/misconfigured server.
	limited := io.LimitReader(resp.Body, 1<<20)
	var body subsResponse
	if err := json.NewDecoder(limited).Decode(&body); err != nil {
		return fmt.Errorf("subscription: decode response: %w", err)
	}

	if len(body.Tunnels) == 0 {
		return errors.New("subscription: empty tunnel list")
	}

	// Inject credentials into each tunnel URL and apply default weight.
	tunnels := make([]Tunnel, len(body.Tunnels))
	for i, t := range body.Tunnels {
		injected, err := injectCreds(t.URL, s.creds)
		if err != nil {
			return fmt.Errorf("subscription: tunnel[%d] URL invalid: %w", i, err)
		}
		w := t.Weight
		if w == 0 && body.Tunnels[i].Weight == 0 {
			// Distinguish "omitted" (default 1) from "explicitly 0" (failover-only).
			// JSON unmarshals missing field as 0. We treat 0 as failover-only
			// in Pick(), but need to detect "omitted" for default=1.
			// Use the raw JSON to check — simpler: just default 0 to 1 when
			// no tunnels have weight set.
		}
		tunnels[i] = Tunnel{URL: injected, Weight: w}
	}

	// Default weight: if all weights are 0, set all to 1 (server omitted weight field).
	allZero := true
	for _, t := range tunnels {
		if t.Weight != 0 {
			allZero = false
			break
		}
	}
	if allZero {
		for i := range tunnels {
			tunnels[i].Weight = 1
		}
	}

	// Determine effective refresh interval.
	interval := s.refresh // client fallback
	if body.Refresh != nil {
		interval = time.Duration(*body.Refresh) * time.Second
	}

	s.mu.Lock()
	s.tunnels = tunnels
	s.fetchedAt = time.Now()
	s.interval = interval
	s.mu.Unlock()

	s.saveCache(tunnels)
	return nil
}

// Tunnels returns the current cached tunnel list (thread-safe).
func (s *Subscription) Tunnels() []Tunnel {
	s.mu.Lock()
	defer s.mu.Unlock()
	result := make([]Tunnel, len(s.tunnels))
	copy(result, s.tunnels)
	return result
}

// injectCreds sets userinfo on a k2v5:// URL to "user:pass".
func injectCreds(rawURL, creds string) (string, error) {
	u, err := url.Parse(rawURL)
	if err != nil {
		return "", err
	}
	user, pass := splitCreds(creds)
	u.User = url.UserPassword(user, pass)
	return u.String(), nil
}

// splitCreds splits "user:pass" into (user, pass).
// Caller guarantees creds contains ':' (validated in NewSubscription).
func splitCreds(creds string) (string, string) {
	user, pass, _ := strings.Cut(creds, ":")
	return user, pass
}

// saveCache writes the tunnel list to disk cache atomically (write-temp + rename).
// Crash during write leaves old cache intact rather than a corrupt partial file.
func (s *Subscription) saveCache(tunnels []Tunnel) {
	if s.cacheFile == "" {
		return
	}
	data, err := json.Marshal(tunnels)
	if err != nil {
		return
	}
	dir := filepath.Dir(s.cacheFile)
	os.MkdirAll(dir, 0700)
	tmp, err := os.CreateTemp(dir, ".subs-*.tmp")
	if err != nil {
		return
	}
	tmpName := tmp.Name()
	if _, err := tmp.Write(data); err != nil {
		tmp.Close()
		os.Remove(tmpName)
		return
	}
	tmp.Close()
	os.Rename(tmpName, s.cacheFile) // atomic on POSIX, best-effort on Windows
}

// loadCache reads the tunnel list from disk cache into memory.
func (s *Subscription) loadCache() {
	if s.cacheFile == "" {
		return
	}
	data, err := os.ReadFile(s.cacheFile)
	if err != nil {
		return
	}
	var tunnels []Tunnel
	if err := json.Unmarshal(data, &tunnels); err != nil {
		return
	}
	if len(tunnels) > 0 {
		s.tunnels = tunnels
	}
}
```

```go
// Pick selects a tunnel using weighted random selection, excluding specified URLs.
// Failover order: (1) weighted random from non-zero weight servers not in exclude,
// (2) weight=0 backup servers not in exclude, (3) error if all excluded.
func (s *Subscription) Pick(exclude []string) (Tunnel, error) {
	s.mu.Lock()
	tunnels := s.tunnels
	s.mu.Unlock()

	if len(tunnels) == 0 {
		return Tunnel{}, errors.New("subscription: no tunnels available")
	}

	excludeSet := make(map[string]bool, len(exclude))
	for _, u := range exclude {
		excludeSet[u] = true
	}

	// Phase 1: weighted random from non-zero weight, non-excluded servers.
	var candidates []Tunnel
	var totalWeight int
	for _, t := range tunnels {
		if t.Weight > 0 && !excludeSet[t.URL] {
			candidates = append(candidates, t)
			totalWeight += t.Weight
		}
	}

	if len(candidates) > 0 {
		r := rand.IntN(totalWeight)
		for _, t := range candidates {
			r -= t.Weight
			if r < 0 {
				return t, nil
			}
		}
		return candidates[len(candidates)-1], nil // shouldn't reach, safety fallback
	}

	// Phase 2: fall back to weight=0 backup servers.
	for _, t := range tunnels {
		if !excludeSet[t.URL] {
			return t, nil
		}
	}

	return Tunnel{}, errors.New("subscription: all tunnels excluded")
}
```

```go
// StartRefresh starts a background goroutine that periodically fetches
// the tunnel list. clientFn is called on each refresh to get the HTTP client
// (allows the caller to switch between direct and proxy).
// Call Close() to stop.
func (s *Subscription) StartRefresh(ctx context.Context, clientFn func() *http.Client) {
	s.mu.Lock()
	interval := s.interval
	s.mu.Unlock()

	if interval <= 0 {
		return // refresh=0 means fetch-only-on-connect
	}

	refreshCtx, cancel := context.WithCancel(ctx)
	s.mu.Lock()
	s.cancel = cancel
	s.done = make(chan struct{})
	s.mu.Unlock()

	go s.refreshLoop(refreshCtx, clientFn)
}

func (s *Subscription) refreshLoop(ctx context.Context, clientFn func() *http.Client) {
	s.mu.Lock()
	done := s.done
	s.mu.Unlock()
	defer close(done)

	for {
		s.mu.Lock()
		interval := s.interval
		s.mu.Unlock()
		if interval <= 0 {
			interval = 1800 * time.Second
		}

		select {
		case <-ctx.Done():
			return
		case <-time.After(interval):
		}

		fetchCtx, fetchCancel := context.WithTimeout(ctx, 30*time.Second)
		s.Fetch(fetchCtx, clientFn())
		fetchCancel()
	}
}

// Close stops background refresh. Idempotent.
func (s *Subscription) Close() {
	s.mu.Lock()
	cancel := s.cancel
	s.cancel = nil
	done := s.done
	s.mu.Unlock()

	if cancel != nil {
		cancel()
	}
	if done != nil {
		<-done
	}
}
```

```go
// resolveSubscriptions scans routes for k2subs:// URLs, fetches tunnel lists,
// picks a server for each, and returns the resolved k2v5:// URLs.
//
// CRITICAL: This does NOT mutate cfg.Routes — the resolved URLs are returned
// separately. The caller builds a resolved copy for engine, but saves the
// ORIGINAL k2subs:// routes in persisted state. Otherwise auto-reconnect
// after crash connects to one fixed server instead of re-resolving.
//
// On error, all already-created Subscription objects are closed (no goroutine leak).
func (d *Daemon) resolveSubscriptions(cfg *config.ClientConfig) (
	subs []*config.Subscription,
	resolved map[int]string, // route index → resolved k2v5:// URL
	err error,
) {
	resolved = make(map[int]string)
	subsCache := filepath.Join(cacheDir(), "subs")

	for i, route := range cfg.Routes {
		if !strings.HasPrefix(route.Via, "k2subs://") {
			continue
		}

		sub, parseErr := config.NewSubscription(route.Via, subsCache)
		if parseErr != nil {
			// Cleanup already-created subs on error.
			for _, s := range subs {
				s.Close()
			}
			return nil, nil, fmt.Errorf("subscription parse: %w", parseErr)
		}

		// Fetch with direct HTTPS (no tunnel established yet).
		fetchCtx, fetchCancel := context.WithTimeout(context.Background(), 15*time.Second)
		fetchErr := sub.Fetch(fetchCtx, &http.Client{
			Timeout: 15 * time.Second,
			Transport: &http.Transport{
				TLSClientConfig: &tls.Config{MinVersion: tls.VersionTLS12},
			},
		})
		fetchCancel()

		if fetchErr != nil {
			// Check disk cache fallback.
			if len(sub.Tunnels()) == 0 {
				for _, s := range subs {
					s.Close()
				}
				return nil, nil, fmt.Errorf("subscription fetch failed (no cache): %w", fetchErr)
			}
			slog.Warn("subscription fetch failed, using cached data", "err", fetchErr)
		}

		tunnel, pickErr := sub.Pick(nil)
		if pickErr != nil {
			for _, s := range subs {
				s.Close()
			}
			return nil, nil, fmt.Errorf("subscription pick: %w", pickErr)
		}

		slog.Info("subscription resolved",
			"from", route.Via[:min(len(route.Via), 40)]+"...",
			"to", config.TruncateVia(tunnel.URL),
			"servers", len(sub.Tunnels()),
		)

		resolved[i] = tunnel.URL
		subs = append(subs, sub)
	}

	return subs, resolved, nil
}

func applyResolvedRoutes(cfg *config.ClientConfig, resolved map[int]string) *config.ClientConfig {

		engineCfg.Routes[i].Via = url
	}
	// Re-derive Server from resolved routes.
	engineCfg.Server = ""
	config.SetDefaults(&engineCfg)
	return &engineCfg
}
```

```go
			return
		}
		refreshSecs := 300
		json.NewEncoder(w).Encode(subsResponse{
			Tunnels: []Tunnel{
				{URL: "k2v5://srv-a:443?ech=abc&pin=sha256:xyz", Weight: 60},
				{URL: "k2v5://srv-b:443?ech=def", Weight: 40},
				{URL: "k2v5://srv-backup:443?ech=ghi", Weight: 0},
			},
			Refresh: &refreshSecs,
		})
	}))
	defer srv.Close()

	subsURL := "k2subs://alice:tok@" + srv.URL[len("https://"):] + "/subs/team?refresh=1800"
	cacheDir := t.TempDir()

	// 1. Parse.
	sub, err := NewSubscription(subsURL, cacheDir)
	if err != nil {
		t.Fatalf("NewSubscription: %v", err)
	}

	// 2. Fetch.
	if err := sub.Fetch(context.Background(), srv.Client()); err != nil {
		t.Fatalf("Fetch: %v", err)
	}

	// 3. Verify tunnels have injected credentials.
	tunnels := sub.Tunnels()
	if len(tunnels) != 3 {
		t.Fatalf("got %d tunnels, want 3", len(tunnels))
	}
	for _, tun := range tunnels {
		if !strings.Contains(tun.URL, "alice:tok@") {
			t.Errorf("credentials not injected in %q", tun.URL)
		}
	}

	// 4. Verify server refresh override.
	sub.mu.Lock()
	interval := sub.interval
	sub.mu.Unlock()
	if interval != 300*time.Second {
		t.Errorf("interval = %v, want 300s", interval)
	}

	// 5. Pick — should never return backup (weight=0) normally.
	picked := map[string]int{}
	for i := 0; i < 1000; i++ {
		tun, err := sub.Pick(nil)
		if err != nil {
			t.Fatal(err)
		}
		picked[tun.URL]++
	}
	for url := range picked {
		if strings.Contains(url, "srv-backup") {
			t.Error("weight=0 server should not be picked normally")
		}
	}

	// 6. Failover — exclude both non-zero, should get backup.
	excludeA := ""
	excludeB := ""
	for _, tun := range tunnels {
		if strings.Contains(tun.URL, "srv-a") {
			excludeA = tun.URL
		}
		if strings.Contains(tun.URL, "srv-b") {
			excludeB = tun.URL
		}
	}
	backup, err := sub.Pick([]string{excludeA, excludeB})
	if err != nil {
		t.Fatalf("Pick with exclusions: %v", err)
	}
	if !strings.Contains(backup.URL, "srv-backup") {
		t.Errorf("expected backup server, got %q", backup.URL)
	}

	// 7. Cache survives restart.
	sub2, _ := NewSubscription(subsURL, cacheDir)
	if len(sub2.Tunnels()) != 3 {
		t.Fatalf("cache not loaded: got %d tunnels", len(sub2.Tunnels()))
	}
}
```
