// BundlesForConfig returns the deduplicated list of k2b file names
// referenced by the given route match configs.
func BundlesForConfig(matches []MatchConfig) []string {
	seen := make(map[string]bool)
	for _, m := range matches {
		if m.All {
			continue
		}
		names := ExpandMatch(m)
		for _, name := range names {
			if file, ok := bundleFileMap[name]; ok {
				seen[file] = true
			}
		}
	}
	out := make([]string, 0, len(seen))
	for f := range seen {
		out = append(out, f)
	}
	return out
}

// Download ensures all named k2b bundles exist in cacheDir.
// For each missing bundle, it races all sources (Happy Eyeballs style)
// and downloads from the fastest. Uses If-Modified-Since for existing files.
// Blocks until all bundles are present or ctx is cancelled.
// Returns nil when all bundles are cached (existing or freshly downloaded).
func Download(ctx context.Context, cacheDir string, bundles []string, sources []string) error {
	if len(bundles) == 0 {
		return nil
	}
	if err := os.MkdirAll(cacheDir, 0o755); err != nil {
		return fmt.Errorf("rule: create cache dir: %w", err)
	}
	if len(sources) == 0 {
		sources = DefaultSources
	}

	for _, name := range bundles {
		if err := downloadOne(ctx, cacheDir, name, sources); err != nil {
			return err
		}
	}
	return nil
}

// downloadOne downloads a single k2b file using Happy Eyeballs across sources.
// If the file already exists, uses If-Modified-Since. On 304, returns immediately.
// On all-fail, retries with exponential backoff until ctx is cancelled.
func downloadOne(ctx context.Context, cacheDir, name string, sources []string) error {
	dest := filepath.Join(cacheDir, name)

	for attempt := 0; ; attempt++ {
		result, err := raceDownload(ctx, dest, name, sources)
		if err == nil {
			if result == downloadCurrent {
				slog.Debug("rule: bundle up to date", "name", name)
			} else {
				slog.Info("rule: bundle downloaded", "name", name)
			}
			return nil
		}

		if ctx.Err() != nil {
			return ctx.Err()
		}

		// If file exists on disk, we can proceed with stale copy.
		if _, statErr := os.Stat(dest); statErr == nil {
			slog.Warn("rule: bundle update failed, using cached", "name", name, "err", err)
			return nil
		}

		// First boot — no cached file. Retry with backoff.
		backoff := backoffDuration(attempt)
		slog.Warn("rule: bundle download failed, retrying",
			"name", name, "err", err, "backoff", backoff)

		select {
		case <-time.After(backoff):
		case <-ctx.Done():
			return ctx.Err()
		}
	}
}

type downloadResult int

const (
	downloadNew     downloadResult = iota // new file written
	downloadCurrent                       // 304 — file is current
)

// raceDownload spawns one goroutine per source, first success wins.
func raceDownload(ctx context.Context, dest, name string, sources []string) (downloadResult, error) {
	type result struct {
		data   []byte
		status downloadResult
		err    error
	}

	ctx, cancel := context.WithTimeout(ctx, 30*time.Second)
	defer cancel()

	ch := make(chan result, len(sources))

	// Get local file mtime for If-Modified-Since.
	var modTime time.Time
	if info, err := os.Stat(dest); err == nil {
		modTime = info.ModTime()
	}

	for _, src := range sources {
		src := src
		go func() {
			data, status, err := fetchBundle(ctx, src+name, modTime)
			ch <- result{data, status, err}
		}()
	}

	var lastErr error
	for range sources {
		r := <-ch
		if r.err != nil {
			lastErr = r.err
			continue
		}
		cancel() // cancel remaining goroutines

		if r.status == downloadCurrent {
			return downloadCurrent, nil
		}

		// Atomic write: tmp → rename.
		tmp := dest + ".tmp"
		if err := os.WriteFile(tmp, r.data, 0o644); err != nil {
			return 0, fmt.Errorf("write tmp: %w", err)
		}
		if err := os.Rename(tmp, dest); err != nil {
			os.Remove(tmp)
			return 0, fmt.Errorf("rename: %w", err)
		}
		return downloadNew, nil
	}
	return 0, fmt.Errorf("all sources failed: %w", lastErr)
}

// fetchBundle downloads a single bundle from one source URL.
func fetchBundle(ctx context.Context, url string, modTime time.Time) ([]byte, downloadResult, error) {
	req, err := http.NewRequestWithContext(ctx, "GET", url, nil)
	if err != nil {
		return nil, 0, err
	}
	if !modTime.IsZero() {
		req.Header.Set("If-Modified-Since", modTime.UTC().Format(http.TimeFormat))
	}

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		return nil, 0, err
	}
	defer resp.Body.Close()

	if resp.StatusCode == http.StatusNotModified {
		return nil, downloadCurrent, nil
	}
	if resp.StatusCode != http.StatusOK {
		return nil, 0, fmt.Errorf("HTTP %d from %s", resp.StatusCode, url)
	}

	data, err := io.ReadAll(resp.Body)
	if err != nil {
		return nil, 0, err
	}
	return data, downloadNew, nil
}

// backoffDuration returns exponential backoff: 1s, 2s, 4s, 8s… cap 60s.
func backoffDuration(attempt int) time.Duration {
	d := time.Second << uint(attempt)
	if d > 60*time.Second {
		d = 60 * time.Second
	}
	return d
}

// Reload atomically replaces the engine's routes and bundles.
// TmpRules are cleared — DNS queries will re-learn them against the new rules.
// Thread-safe: callers of Match/MatchConn block briefly during swap.
//
// Concurrency: builds new data outside lock, swaps under WLock, closes old
// bundles outside lock (Rule #4: no Close under state lock).
func (e *Engine) Reload(routes []RouteEntry, bundles []*Bundle) error {
	// Swap under write lock.
	e.mu.Lock()
	oldBundles := e.bundles
	e.bundles = bundles
	e.routes = routes
	e.tmp = make(map[netip.Addr]Target) // clear TmpRules
	e.mu.Unlock()

	// Close old bundles outside lock.
	_ = oldBundles // Bundle has no explicit Close — data is GC'd.
	return nil
}

	sources []string,

// UpdaterOption configures the updater.
type UpdaterOption func(*updaterConfig)

type updaterConfig struct {
	interval time.Duration
}

// WithCheckInterval overrides the default 24h check interval (for testing).

	return func(c *updaterConfig) {
		c.interval = d
	}

// RebuildFunc loads bundles from cacheDir and rebuilds the engine's routes.
// Called by the updater after downloading new bundles.
// Returns the new routes and bundles for Engine.Reload().
type RebuildFunc func(cacheDir string) ([]RouteEntry, []*Bundle, error)

// StartUpdater launches a background goroutine that periodically checks for
// bundle updates. When new bundles are downloaded, it calls rebuild to get
// new routes and bundles, then swaps them into the engine via Reload().
// Stops when ctx is cancelled.
func StartUpdater(
	ctx context.Context,
	engine *Engine,
	cacheDir string,
	bundles []string,
	sources []string,
	rebuild RebuildFunc,
	opts ...UpdaterOption,
) {
	cfg := updaterConfig{interval: defaultCheckInterval}
	for _, o := range opts {
		o(&cfg)
	}

	if len(bundles) == 0 {
		return
	}

	go func() {
		ticker := time.NewTicker(cfg.interval)
		defer ticker.Stop()

		for {
			select {
			case <-ctx.Done():
				return
			case <-ticker.C:
				if err := Download(ctx, cacheDir, bundles, sources); err != nil {
					slog.Warn("rule: updater download failed", "err", err)
					continue
				}
				routes, newBundles, err := rebuild(cacheDir)
				if err != nil {
					slog.Warn("rule: updater rebuild failed", "err", err)
					continue
				}
				if err := engine.Reload(routes, newBundles); err != nil {
					slog.Warn("rule: updater reload failed", "err", err)
					continue
				}
				slog.Info("rule: updater reload complete", "routes", len(routes))
			}
		}
	}()
}

// buildRouteEntries converts config routes to rule.RouteEntry list using the given bundle index.
// Shared by buildRuleEngine (startup) and updater rebuild (hot-reload) — single code path
// ensures both always produce identical routing tables from the same config+bundles.
func buildRouteEntries(routes []config.RouteConfig, idx map[string]*rule.BundleSet) (entries []rule.RouteEntry, fallback rule.Target, allProxy bool) {
	fallback = rule.TargetDirect

	for i, route := range routes {
		target := rule.ParseVia(route.Via)

		if route.Match.All {
			allProxy = (target == rule.TargetProxy)
			fallback = target
			break
		}

		entry := rule.RouteEntry{Target: target}

		// Bundle-based sets.
		names := rule.ExpandMatch(route.Match)
		for _, name := range names {
			if bs, ok := idx[name]; ok {
				entry.Sets = append(entry.Sets, bs)
			}
		}

		// Inline rules.
		if len(route.Match.DomainSuffix) > 0 || len(route.Match.IPCIDR) > 0 {
			inline := rule.BuildInlineSet(route.Match.DomainSuffix, route.Match.IPCIDR)
			entry.Inline = &inline
		}

		// Meta criteria.
		entry.ProcessNames = route.Match.ProcessName
		entry.PackageNames = route.Match.PackageName
		entry.NetworkMatch = route.Match.Network
		entry.IPIsPrivate = route.Match.IPIsPrivate

		hasAnyCriteria := len(entry.Sets) > 0 ||
			(entry.Inline != nil && !entry.Inline.IsEmpty()) ||
			len(entry.ProcessNames) > 0 || len(entry.PackageNames) > 0 ||
			entry.NetworkMatch != "" || entry.IPIsPrivate
		if hasAnyCriteria {
			entries = append(entries, entry)
		}

		// Last route without match criteria = fallback.
		if i == len(routes)-1 && !route.Match.HasCriteria() {
			fallback = target
		}
	}
	return entries, fallback, allProxy
}
